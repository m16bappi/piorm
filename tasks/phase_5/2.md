## RouterChain (Rust)

```rust
// ormpi-core/src/routing.rs
use std::sync::Arc;
use arc_swap::ArcSwap;
use once_cell::sync::Lazy;

static ROUTER_CHAIN: Lazy<ArcSwap<RouterChain>> =
    Lazy::new(|| ArcSwap::from_pointee(RouterChain::default()));

pub struct RouterChain {
    routers: Vec<PyObject>,   // Python router instances, stored as Arc<Mutex<>>
}

impl RouterChain {
    pub fn resolve_read(&self, model_name: &str) -> String {
        Python::with_gil(|py| {
            for router in &self.routers {
                let result = router.call_method1(
                    py, "db_for_read",
                    (model_name,),
                ).ok();

                if let Some(alias) = result.and_then(|r| r.extract::<Option<String>>(py).ok()).flatten() {
                    return alias;
                }
            }
            "default".to_string()
        })
    }

    pub fn resolve_write(&self, model_name: &str) -> String {
        Python::with_gil(|py| {
            for router in &self.routers {
                let result = router.call_method1(
                    py, "db_for_write",
                    (model_name,),
                ).ok();

                if let Some(alias) = result.and_then(|r| r.extract::<Option<String>>(py).ok()).flatten() {
                    return alias;
                }
            }
            "default".to_string()
        })
    }

    pub fn allow_migrate(&self, db: &str, model_name: &str) -> bool {
        Python::with_gil(|py| {
            for router in &self.routers {
                let result = router.call_method1(
                    py, "allow_migrate",
                    (db, model_name),
                ).ok();

                if let Some(allowed) = result.and_then(|r| r.extract::<Option<bool>>(py).ok()).flatten() {
                    return allowed;
                }
            }
            true  // default: allow all migrations
        })
    }
}

pub fn set_routers(router_instances: Vec<PyObject>) {
    ROUTER_CHAIN.store(Arc::new(RouterChain { routers: router_instances }));
}

pub fn get_read_db(model_name: &str, explicit_using: Option<&str>) -> String {
    if let Some(alias) = explicit_using {
        return alias.to_string();
    }
    ROUTER_CHAIN.load().resolve_read(model_name)
}

pub fn get_write_db(model_name: &str, explicit_using: Option<&str>) -> String {
    if let Some(alias) = explicit_using {
        return alias.to_string();
    }
    ROUTER_CHAIN.load().resolve_write(model_name)
}
```

## Router Loading from Paths

When `configure(database_routers=["myapp.routers.PrimaryReplicaRouter"])` is called:

```rust
pub fn load_routers(py: Python<'_>, router_paths: &[String]) -> PyResult<Vec<PyObject>> {
    router_paths.iter().map(|path| {
        // "myapp.routers.PrimaryReplicaRouter" → import myapp.routers, get PrimaryReplicaRouter
        let parts: Vec<&str> = path.rsplitn(2, '.').collect();
        let (class_name, module_path) = (parts[0], parts[1]);
        let module = py.import(module_path)?;
        let cls = module.getattr(class_name)?;
        cls.call0()  // instantiate the router
    }).collect()
}
```

## DatabaseRouter ABC (Python-side)

```python
# python/ormpi/routers.py
class DatabaseRouter:
    """
    Abstract base class for database routers.
    All methods return None to defer to the next router.
    """

    def db_for_read(self, model, **hints):
        """Return the database alias to use for reading `model`."""
        return None

    def db_for_write(self, model, **hints):
        """Return the database alias to use for writing `model`."""
        return None

    def allow_relation(self, obj1, obj2, **hints):
        """Return True if a relation between obj1 and obj2 is allowed."""
        return None

    def allow_migrate(self, db, app_label, model_name=None, **hints):
        """Return True if migration is allowed on `db` for this model."""
        return None
```

## Example Router

```python
# Example: Read from replica, write to default
class PrimaryReplicaRouter:
    def db_for_read(self, model, **hints):
        return "replica"

    def db_for_write(self, model, **hints):
        return "default"

    def allow_migrate(self, db, app_label, model_name=None, **hints):
        return db == "default"   # only run migrations on primary
```

## Acceptance Criteria
- [ ] `PrimaryReplicaRouter` routes reads to "replica" and writes to "default"
- [ ] Router chain: first non-None result wins, falls back to "default"
- [ ] `using("alias")` always overrides the router
- [ ] Router `allow_migrate` respected by migration executor
- [ ] Loading router from dotted path string works
- [ ] Multiple routers in chain evaluated in order
- [ ] Router returning `None` defers to next router
- [ ] Empty router list → always uses "default"
