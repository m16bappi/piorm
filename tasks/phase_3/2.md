```rust
#[pymethods]
impl PyModel {
    #[new]
    #[pyo3(signature = (**kwargs))]
    fn __init__(slf: PyRefMut<Self>, py: Python<'_>, kwargs: Option<&PyDict>) -> PyResult<()> {
        // Create _ormpi_values dict (stores field values)
        let values = PyDict::new(py);
        // Create _ormpi_dirty set (tracks which fields changed)
        let dirty = PySet::empty(py)?;
        // Set _state
        let state = PyDict::new(py);
        state.set_item("adding", true)?;    // True = new instance (INSERT)
        state.set_item("db", py.None())?;

        slf.as_ref().setattr("_ormpi_values", values)?;
        slf.as_ref().setattr("_ormpi_dirty", dirty)?;
        slf.as_ref().setattr("_ormpi_state", state)?;

        // Apply kwargs to field values
        if let Some(kw) = kwargs {
            for (key, value) in kw.iter() {
                let field_name: String = key.extract()?;
                slf.as_ref().setattr(field_name.as_str(), value)?;  // goes through descriptor
            }
        }

        Ok(())
    }

    #[getter]
    fn pk(&self, py: Python<'_>, slf: &PyAny) -> PyResult<PyObject> {
        // Look up the pk field name from registry, then get its value
        let model_def = get_model_def_for_instance(py, slf)?;
        let pk_name = &model_def.pk_field;
        let values = slf.getattr("_ormpi_values")?;
        Ok(values.get_item(pk_name).unwrap_or(py.None()).into())
    }

    #[setter]
    fn set_pk(&self, slf: &PyAny, value: PyObject) -> PyResult<()> {
        let py = value.py();
        let model_def = get_model_def_for_instance(py, slf)?;
        let pk_name = &model_def.pk_field;
        let values = slf.getattr("_ormpi_values")?;
        values.set_item(pk_name, value)?;
        Ok(())
    }
}
```

## save() — Rust Core Logic

```rust
// ormpi-core/src/model/executor.rs

pub async fn save_instance(
    pool: &OrmpiPool,
    backend: QueryBackend,
    model_def: &ModelDef,
    values: HashMap<String, OrmpiValue>,
    dirty_fields: HashSet<String>,
    is_adding: bool,
    force_insert: bool,
    force_update: bool,
    update_fields: Option<Vec<String>>,
) -> Result<OrmpiValue> {   // Returns the pk value

    let table = sea_query::Alias::new(&model_def.table_name);

    if is_adding || force_insert {
        // INSERT
        let mut stmt = sea_query::Query::insert().into_table(table).to_owned();

        let fields_to_insert: Vec<&FieldDef> = model_def.fields.values()
            .filter(|f| !f.primary_key || values.contains_key(&f.attname))
            .filter(|f| !matches!(f.field_type, FieldType::ManyToMany { .. }))
            .collect();

        stmt.columns(fields_to_insert.iter().map(|f| sea_query::Alias::new(&f.column_name())));

        let row_values: Vec<sea_query::Expr> = fields_to_insert.iter()
            .map(|f| {
                let val = values.get(&f.attname).cloned().unwrap_or(OrmpiValue::Null);
                sea_query::Expr::val(val.to_sea_value())
            })
            .collect();
        stmt.values_panic(row_values);

        // For Postgres, add RETURNING id to get the inserted pk
        if matches!(backend.0, DatabaseBackend::Postgres) {
            stmt.returning_col(sea_query::Alias::new(&model_def.pk_field));
        }

        let (sql, sea_values) = backend.build_insert(&stmt);
        let pk = execute_insert(pool, &sql, &sea_values, model_def).await?;
        Ok(pk)

    } else {
        // UPDATE
        let fields_to_update: Vec<&FieldDef> = model_def.fields.values()
            .filter(|f| {
                let name = &f.attname;
                if let Some(ref uf) = update_fields {
                    uf.contains(name)
                } else {
                    dirty_fields.contains(name) && !f.primary_key
                }
            })
            .collect();

        if fields_to_update.is_empty() {
            // Nothing to update
            return Ok(values.get(&model_def.pk_field).cloned().unwrap_or(OrmpiValue::Null));
        }

        let mut stmt = sea_query::Query::update().table(table).to_owned();

        for field in &fields_to_update {
            let val = values.get(&field.attname).cloned().unwrap_or(OrmpiValue::Null);
            stmt.value(sea_query::Alias::new(&field.column_name()), val.to_sea_value());
        }

        let pk_val = values.get(&model_def.pk_field)
            .ok_or(OrmpiError::FieldError {
                model: model_def.model_name.clone(),
                field: model_def.pk_field.clone(),
                message: "pk value is None, cannot UPDATE".to_string(),
            })?;

        stmt.and_where(
            sea_query::Expr::col(sea_query::Alias::new(&model_def.pk_field))
                .eq(pk_val.to_sea_value())
        );

        let (sql, sea_values) = backend.build_update(&stmt);
        execute_update(pool, &sql, &sea_values).await?;
        Ok(pk_val.clone())
    }
}
```

## save() — PyO3 Wrapper

```rust
#[pymethods]
impl PyModel {
    fn save<'py>(
        slf: &'py PyAny,
        py: Python<'py>,
        using: Option<String>,
        force_insert: bool,
        force_update: bool,
        update_fields: Option<Vec<String>>,
    ) -> PyResult<&'py PyAny> {
        let using = using.unwrap_or_else(|| "default".to_string());

        // Collect values and dirty set from Python
        let values_dict = slf.getattr("_ormpi_values")?;
        let dirty_set = slf.getattr("_ormpi_dirty")?;
        let state = slf.getattr("_ormpi_state")?;
        let is_adding: bool = state.get_item("adding")?.extract()?;

        // Convert Python dict to HashMap<String, OrmpiValue>
        let values = py_dict_to_values(py, values_dict)?;
        let dirty_fields: HashSet<String> = dirty_set.iter()?.map(|k| k?.extract()).collect::<PyResult<_>>()?;

        // Get model def
        let model_def = get_model_def_for_instance(py, slf)?.clone();

        let slf_ref = slf.into_py(py);
        pyo3_async_runtimes::tokio::future_into_py(py, async move {
            let registry = DatabaseRegistry::global();
            let reg = registry.read().unwrap();
            let pool = reg.get_pool(&using)?;
            let backend = QueryBackend(pool.backend());

            let pk = save_instance(
                pool, backend, &model_def, values, dirty_fields,
                is_adding, force_insert, force_update, update_fields,
            ).await?;

            Python::with_gil(|py| {
                let instance = slf_ref.as_ref(py);
                // Set pk
                let values_dict = instance.getattr("_ormpi_values")?;
                values_dict.set_item(&model_def.pk_field, value_to_py(py, &pk))?;
                // Clear dirty set
                instance.getattr("_ormpi_dirty")?.call_method0("clear")?;
                // Set adding=False
                let state = instance.getattr("_ormpi_state")?;
                state.set_item("adding", false)?;
                state.set_item("db", &using)?;
                Ok(py.None())
            })
        })
    }
```

```rust
pub async fn delete_instance(
    pool: &OrmpiPool,
    backend: QueryBackend,
    model_def: &ModelDef,
    pk_value: OrmpiValue,
    using: &str,
) -> Result<(usize, HashMap<String, usize>)> {
    let table = sea_query::Alias::new(&model_def.table_name);
    let pk_col = sea_query::Alias::new(&model_def.pk_field);

    // TODO Phase 4: handle CASCADE deletes

    let stmt = sea_query::Query::delete()
        .from_table(table)
        .and_where(sea_query::Expr::col(pk_col).eq(pk_value.to_sea_value()))
        .to_owned();

    let (sql, values) = backend.build_delete(&stmt);
    let rows_affected = execute_delete(pool, &sql, &values).await?;

    let mut counts = HashMap::new();
    counts.insert(model_def.model_name.clone(), rows_affected);
    Ok((rows_affected, counts))
}
```
