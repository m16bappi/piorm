# Task 3.1 — ModelDef, ModelRegistry & __init_subclass__ Hook

## Objective
Implement the Rust `ModelDef` and `ModelRegistry`, and wire up the Python `Model.__init_subclass__` hook so that defining a model class automatically registers it in Rust. This is the core metaclass-like mechanism.

## Location
- `ormpi-core/src/model/def.rs`
- `ormpi-core/src/model/registry.rs`
- `ormpi-py/src/model/base.rs`

## ModelDef (Rust)

```rust
// ormpi-core/src/model/def.rs
use indexmap::IndexMap;
use crate::fields::FieldDef;

#[derive(Debug, Clone)]
pub struct ModelDef {
    pub app_label: String,
    pub model_name: String,           // PascalCase class name
    pub table_name: String,           // snake_case table name
    pub fields: IndexMap<String, FieldDef>,  // ordered, keyed by attname
    pub pk_field: String,             // name of the PK field
    pub ordering: Vec<OrderSpec>,
    pub unique_together: Vec<Vec<String>>,
    pub index_together: Vec<Vec<String>>,  // deprecated but supported
    pub indexes: Vec<IndexDef>,
    pub constraints: Vec<ConstraintDef>,
    pub m2m_fields: Vec<M2MFieldDef>,
    pub abstract_model: bool,
    pub proxy: bool,
    pub managed: bool,                // if False, ormpi won't create/migrate the table
    pub default_manager_name: String,
}

#[derive(Debug, Clone)]
pub struct OrderSpec {
    pub field: String,
    pub descending: bool,
    pub nulls_last: bool,
    pub nulls_first: bool,
}

#[derive(Debug, Clone)]
pub struct IndexDef {
    pub name: Option<String>,
    pub fields: Vec<String>,
    pub unique: bool,
    pub condition: Option<String>,    // partial index WHERE clause
}

#[derive(Debug, Clone)]
pub struct ConstraintDef {
    pub name: String,
    pub kind: ConstraintKind,
}

#[derive(Debug, Clone)]
pub enum ConstraintKind {
    UniqueConstraint { fields: Vec<String>, condition: Option<String> },
    CheckConstraint { check: String },
}

#[derive(Debug, Clone)]
pub struct M2MFieldDef {
    pub field_name: String,         // attribute name on the model
    pub to_model: String,
    pub through_model: Option<String>,
    pub junction_table: String,     // computed: "{from_model}_{field_name}"
    pub from_column: String,        // "{from_model_lower}_id"
    pub to_column: String,          // "{to_model_lower}_id"
}

impl ModelDef {
    /// Compute the default table name from app_label and model_name.
    pub fn default_table_name(app_label: &str, model_name: &str) -> String {
        format!("{}_{}", app_label, model_name.to_lowercase())
    }

    /// Get the primary key FieldDef.
    pub fn pk(&self) -> Option<&FieldDef> {
        self.fields.get(&self.pk_field)
    }
}
```

## ModelRegistry (Rust)

```rust
// ormpi-core/src/model/registry.rs
use std::collections::HashMap;
use std::sync::{Arc, RwLock};
use once_cell::sync::Lazy;

static MODEL_REGISTRY: Lazy<Arc<RwLock<ModelRegistry>>> =
    Lazy::new(|| Arc::new(RwLock::new(ModelRegistry::new())));

pub struct ModelRegistry {
    models: HashMap<String, ModelDef>,   // key: "app_label.ModelName"
}

impl ModelRegistry {
    pub fn new() -> Self { Self { models: HashMap::new() } }

    pub fn global() -> Arc<RwLock<ModelRegistry>> {
        MODEL_REGISTRY.clone()
    }

    pub fn register(&mut self, def: ModelDef) {
        let key = format!("{}.{}", def.app_label, def.model_name);
        self.models.insert(key, def);
    }

    pub fn get(&self, app_label: &str, model_name: &str) -> Option<&ModelDef> {
        let key = format!("{}.{}", app_label, model_name);
        self.models.get(&key)
    }

    pub fn get_by_name(&self, model_name: &str) -> Option<&ModelDef> {
        self.models.values().find(|m| m.model_name == model_name)
    }

    pub fn all(&self) -> impl Iterator<Item = &ModelDef> {
        self.models.values()
    }

    pub fn get_all_for_app(&self, app_label: &str) -> Vec<&ModelDef> {
        self.models.values().filter(|m| m.app_label == app_label).collect()
    }
}
```

## Python Model Base Class

```rust
// ormpi-py/src/model/base.rs
use pyo3::prelude::*;

#[pyclass(name = "Model", subclass)]
pub struct PyModel {
    // Shared state — actual data is in Python __dict__ via _ormpi_values
}

#[pymethods]
impl PyModel {
    /// Called automatically when a subclass is created.
    /// This is the metaclass hook.
    #[classmethod]
    fn __init_subclass__(
        cls: &PyType,
        py: Python<'_>,
        #[pyo3(from_py_with = "extract_kwargs")] kwargs: HashMap<String, PyObject>,
    ) -> PyResult<()> {
        // Skip abstract classes
        let meta = cls.getattr("Meta").ok();
        let is_abstract = meta.as_ref()
            .and_then(|m| m.getattr("abstract").ok())
            .and_then(|v| v.extract::<bool>().ok())
            .unwrap_or(false);

        if is_abstract { return Ok(()); }

        // Collect fields from class __dict__
        let class_dict = cls.getattr("__dict__")?;
        let mut fields = IndexMap::new();
        let mut m2m_fields = Vec::new();
        let mut has_pk = false;

        for (key, value) in class_dict.items()? {
            let key: String = key.extract()?;
            if key.starts_with('_') { continue; }

            // Check if value is a field class
            if let Ok(field) = value.downcast::<PyCharField>() { ... }
            // ... check all field types

            if let Some(mut field_def) = extract_field_def(py, &value)? {
                // Call contribute_to_class
                value.call_method1("contribute_to_class", (cls, &key))?;

                if matches!(field_def.field_type, FieldType::ManyToMany { .. }) {
                    m2m_fields.push(extract_m2m_def(&field_def, cls)?);
                } else {
                    if field_def.primary_key { has_pk = true; }
                    fields.insert(field_def.attname.clone(), field_def);
                }
            }
        }

        // Auto-add BigAutoField pk if no explicit pk
        if !has_pk {
            let auto_pk = FieldDef {
                name: "id".to_string(),
                attname: "id".to_string(),
                field_type: FieldType::AutoBigInt,
                primary_key: true,
                ..Default::default()
            };
            fields.insert("id".to_string(), auto_pk);
        }

        // Parse Meta class
        let (app_label, db_table, ordering, unique_together, indexes) = parse_meta(cls, meta.as_deref())?;

        // Build ModelDef
        let class_name = cls.name()?.to_string();
        let model_def = ModelDef {
            app_label: app_label.clone(),
            model_name: class_name.clone(),
            table_name: db_table.unwrap_or_else(|| ModelDef::default_table_name(&app_label, &class_name)),
            fields,
            pk_field: "id".to_string(),
            ordering,
            unique_together,
            index_together: vec![],
            indexes,
            constraints: vec![],
            m2m_fields,
            abstract_model: false,
            proxy: false,
            managed: true,
            default_manager_name: "objects".to_string(),
        };

        // Register in global registry
        let registry = ModelRegistry::global();
        registry.write().unwrap().register(model_def);

        // Attach Manager as `cls.objects`
        let manager = PyManager::new(app_label, class_name);
        cls.setattr("objects", Py::new(py, manager)?)?;

        // Attach DoesNotExist exception
        let does_not_exist = py.import("ormpi.exceptions")?
            .getattr("DoesNotExist")?;
        cls.setattr("DoesNotExist", does_not_exist)?;

        Ok(())
    }
```

## Meta Parsing

```rust
fn parse_meta(
    cls: &PyType,
    meta: Option<&PyAny>,
) -> PyResult<(String, Option<String>, Vec<OrderSpec>, Vec<Vec<String>>, Vec<IndexDef>)> {
    let app_label = meta
        .and_then(|m| m.getattr("app_label").ok())
        .and_then(|v| v.extract::<String>().ok())
        .unwrap_or_else(|| infer_app_label(cls));

    let db_table = meta
        .and_then(|m| m.getattr("db_table").ok())
        .and_then(|v| v.extract::<String>().ok());

    let ordering = meta
        .and_then(|m| m.getattr("ordering").ok())
        .and_then(|v| v.extract::<Vec<String>>().ok())
        .unwrap_or_default()
        .into_iter()
        .map(|s| parse_order_spec(&s))
        .collect();

    // ... parse unique_together, indexes, etc.
    Ok((app_label, db_table, ordering, vec![], vec![]))
}

fn infer_app_label(cls: &PyType) -> String {
    // Use the module name: "myapp.models" → "myapp"
    cls.getattr("__module__")
        .and_then(|m| m.extract::<String>())
        .map(|m| m.split('.').next().unwrap_or("default").to_string())
        .unwrap_or_else(|_| "default".to_string())
}
```
