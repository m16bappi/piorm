# Task 3.3 — Manager Class & DDL (CREATE TABLE from ModelDef)

## Objective
Implement the `Manager` class (bound to `Model.objects`) and the DDL generation function that converts a `ModelDef` to a `CREATE TABLE` statement, including indexes, unique constraints, and foreign keys. This DDL function is the foundation for the migration engine.

## Location
- `ormpi-py/src/model/manager.rs`
- `ormpi-core/src/model/ddl.rs`

```rust
// ormpi-py/src/model/manager.rs
#[pyclass(name = "Manager")]
pub struct PyManager {
    pub app_label: String,
    pub model_name: String,
}

#[pymethods]
impl PyManager {
    pub fn get_queryset(&self, py: Python<'_>) -> PyResult<PyObject> {
        // Returns QuerySet(app_label, model_name)
        // Full QuerySet in Phase 4 — for now return a stub
        let qs = PyQuerySet::new(self.app_label.clone(), self.model_name.clone());
        Ok(Py::new(py, qs)?.into_py(py))
    }

    // Forward all standard QuerySet methods
    pub fn all(&self, py: Python<'_>) -> PyResult<PyObject> {
        self.get_queryset(py)?.call_method0(py, "all")
    }

    pub fn filter(&self, py: Python<'_>, args: &PyTuple, kwargs: Option<&PyDict>) -> PyResult<PyObject> {
        self.get_queryset(py)?.call_method(py, "filter", args, kwargs)
    }

    pub fn exclude(&self, py: Python<'_>, args: &PyTuple, kwargs: Option<&PyDict>) -> PyResult<PyObject> {
        self.get_queryset(py)?.call_method(py, "exclude", args, kwargs)
    }

    pub fn get<'py>(&self, py: Python<'py>, args: &PyTuple, kwargs: Option<&PyDict>) -> PyResult<&'py PyAny> {
        let qs = self.get_queryset(py)?;
        qs.call_method(py, "get", args, kwargs)?.as_ref(py).call0()
    }

    pub fn create<'py>(&self, py: Python<'py>, kwargs: Option<&PyDict>) -> PyResult<&'py PyAny> {
        let qs = self.get_queryset(py)?;
        qs.call_method(py, "create", PyTuple::empty(py), kwargs)?.as_ref(py).call0()
    }

    pub fn bulk_create<'py>(&self, py: Python<'py>, objs: Vec<PyObject>, batch_size: Option<usize>) -> PyResult<&'py PyAny> {
        let qs = self.get_queryset(py)?;
        let args = PyTuple::new(py, [objs.into_py(py)]);
        qs.call_method1(py, "bulk_create", args)?.as_ref(py).call0()
    }

    pub fn __repr__(&self) -> String {
        format!("<Manager for {}.{}>", self.app_label, self.model_name)
    }
}
```

## DDL Generation

```rust
// ormpi-core/src/model/ddl.rs
use sea_query::{Table, ColumnDef, ForeignKey, ForeignKeyAction, Index};

pub struct DdlStatements {
    pub create_table: String,
    pub create_indexes: Vec<String>,
    pub create_m2m_tables: Vec<String>,
}

pub fn model_to_ddl(model_def: &ModelDef, backend: QueryBackend) -> DdlStatements {
    // 1. BUILD CREATE TABLE
    let mut table = Table::create()
        .table(sea_query::Alias::new(&model_def.table_name))
        .if_not_exists()
        .to_owned();

    // Add columns
    for field in model_def.fields.values() {
        if let Some(col_def) = field.to_column_def() {
            table.col(col_def);
        }
    }

    // Add unique_together constraints
    for unique_group in &model_def.unique_together {
        let constraint = sea_query::Index::create()
            .unique()
            .col_tpl(unique_group.iter().map(|c| sea_query::Alias::new(c)));
        // Add as table constraint
    }

    // Add foreign key constraints (after all columns)
    for field in model_def.fields.values() {
        if let FieldType::ForeignKey { to, on_delete, .. } = &field.field_type {
            let ref_model = resolve_model_name(to);
            let fk = sea_query::ForeignKey::create()
                .from(
                    sea_query::Alias::new(&model_def.table_name),
                    sea_query::Alias::new(&field.column_name()),
                )
                .to(
                    sea_query::Alias::new(&ref_model.table_name),
                    sea_query::Alias::new(&ref_model.pk_field),
                )
                .on_delete(on_delete_to_sea(*on_delete))
                .on_update(ForeignKeyAction::NoAction)
                .to_owned();
            table.foreign_key(&fk);
        }
    }

    let create_sql = backend.build_table_create(&table);

    // 2. BUILD CREATE INDEX statements
    let mut create_indexes = Vec::new();

    // db_index=True fields
    for field in model_def.fields.values() {
        if field.db_index && !field.unique && !field.primary_key {
            let idx = Index::create()
                .name(&format!("idx_{}_{}", model_def.table_name, field.column_name()))
                .table(sea_query::Alias::new(&model_def.table_name))
                .col(sea_query::Alias::new(&field.column_name()))
                .to_owned();
            create_indexes.push(backend.build_index_create(&idx));
        }
    }

    // Meta.indexes
    for index_def in &model_def.indexes {
        let name = index_def.name.clone().unwrap_or_else(|| {
            format!("idx_{}_{}", model_def.table_name, index_def.fields.join("_"))
        });
        let mut idx = Index::create()
            .name(&name)
            .table(sea_query::Alias::new(&model_def.table_name))
            .to_owned();
        for col in &index_def.fields {
            idx.col(sea_query::Alias::new(col));
        }
        if index_def.unique { idx.unique(); }
        create_indexes.push(backend.build_index_create(&idx));
    }

    // 3. BUILD M2M JUNCTION TABLES
    let mut create_m2m_tables = Vec::new();
    for m2m in &model_def.m2m_fields {
        if m2m.through_model.is_none() {
            // Auto-generated junction table
            let mut jt = Table::create()
                .table(sea_query::Alias::new(&m2m.junction_table))
                .if_not_exists()
                .col(ColumnDef::new(sea_query::Alias::new("id")).big_integer().auto_increment().primary_key())
                .col(ColumnDef::new(sea_query::Alias::new(&m2m.from_column)).big_integer().not_null())
                .col(ColumnDef::new(sea_query::Alias::new(&m2m.to_column)).big_integer().not_null())
                .to_owned();
            create_m2m_tables.push(backend.build_table_create(&jt));
        }
    }

    DdlStatements { create_table: create_sql, create_indexes, create_m2m_tables }
}

fn on_delete_to_sea(action: OnDelete) -> ForeignKeyAction {
    match action {
        OnDelete::Cascade    => ForeignKeyAction::Cascade,
        OnDelete::SetNull    => ForeignKeyAction::SetNull,
        OnDelete::SetDefault => ForeignKeyAction::SetDefault,
        OnDelete::Restrict   => ForeignKeyAction::Restrict,
        OnDelete::DoNothing  | OnDelete::Protect => ForeignKeyAction::NoAction,
    }
}
```
