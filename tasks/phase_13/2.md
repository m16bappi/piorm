# Task 13.2 — Self-Typed Model API (`models.pyi`, `_core.pyi`, `__init__.pyi`)

## Objective

Three stubs need updating to eliminate `Any` from the model layer:

1. **`models.pyi`** — apply `@dataclass_transform`, change `objects: ClassVar[Manager[Self]]`,
   fix `from_db` to return `Self`.
2. **`_core.pyi`** — update the minimal `Model` forward-declaration stub; fix field stubs to
   reference `Field[T]` from `fields.pyi` rather than bare `Field`.
3. **`__init__.pyi`** — re-export `Manager` and `QuerySet` so users can write
   `from ormpi import Manager` for custom manager annotations.

## `python/ormpi/models.pyi` — full rewrite

```python
# python/ormpi/models.pyi
# Python 3.12+ stubs for the ormpi Model base class.

from typing import Any, ClassVar, Self, dataclass_transform
from ormpi._core import Manager, QuerySet
from ormpi.exceptions import DoesNotExist as _DoesNotExist
from ormpi.fields import Field

# ── ModelOptions ──────────────────────────────────────────────────────────

class ModelOptions:
    """Mirrors the ``class Meta`` inner class of a model."""
    db_table:        str
    ordering:        list[str]
    unique_together: list[list[str]]
    managed:         bool
    abstract:        bool

# ── Model ─────────────────────────────────────────────────────────────────

@dataclass_transform(
    eq_default=True,
    kw_only_default=True,
    field_specifiers=(Field,),
)
class Model:
    """Base class for all ormpi models.

    Subclass to define a table::

        class Author(Model):
            name  = CharField(max_length=100)
            email = CharField(max_length=200, unique=True)
            bio   = TextField(null=True)

    Type checkers that understand ``@dataclass_transform`` synthesise a typed
    ``__init__`` from the declared fields::

        Author(name="Alice", email="a@example.com", bio=None)
        # (name: str, email: str, bio: str | None)

    ``objects`` is resolved per-subclass via ``Self``::

        Author.objects           # Manager[Author]
        Author.objects.get(pk=1) # Coroutine[Any, Any, Author]
    """

    # ── Private ORM state ──────────────────────────────────────────────
    _ormpi_values: dict[str, Any]
    _ormpi_dirty:  set[str]
    _ormpi_state:  dict[str, Any]

    # ── Class-level attributes ─────────────────────────────────────────
    #
    # ``objects`` typed as ``Manager[Self]`` so that:
    #   Author.objects.get(pk=1)   →  Coroutine[Any, Any, Author]
    #   Author.objects.filter(...) →  QuerySet[Author]
    #   Author.objects.first()     →  Coroutine[Any, Any, Author | None]
    #
    # Pyright resolves Self correctly when accessed on a concrete subclass.
    # Mypy note: mypy < 1.12 has partial Self-in-ClassVar support. If mypy
    # strict mode rejects ``Manager[Self]`` here, use:
    #   objects: ClassVar[Manager[Any]]
    # as a temporary workaround — pyright will still infer correctly.
    objects:      ClassVar[Manager[Self]]
    DoesNotExist: ClassVar[type[_DoesNotExist]]
    _meta:        ClassVar[ModelOptions]

    # ── Primary key ────────────────────────────────────────────────────
    @property
    def pk(self) -> Any:
        """Shorthand for the primary-key field value.

        The type is ``Any`` because the PK type depends on which field the
        user declared (or the implicit BIGINT if none).  Annotate your model's
        ``id`` field explicitly for a narrower type.
        """
        ...

    # ── Constructor ────────────────────────────────────────────────────
    #
    # ``@dataclass_transform`` synthesises a typed __init__ per subclass.
    # The fallback below is used for Model itself (not concrete subclasses).
    def __init__(self, **kwargs: Any) -> None: ...

    def __repr__(self) -> str: ...
    def __eq__(self, other: object) -> bool: ...
    def __hash__(self) -> int: ...

    # ── Persistence ────────────────────────────────────────────────────
    async def save(
        self,
        update_fields: list[str] | None = None,
        using: str = "default",
    ) -> None: ...

    async def delete(self, using: str = "default") -> None: ...

    async def refresh_from_db(
        self,
        fields: list[str] | None = None,
        using: str = "default",
    ) -> None: ...

    # ── Class methods ──────────────────────────────────────────────────
    @classmethod
    def from_db(
        cls,
        db: str,
        field_names: list[str],
        values: list[Any],
    ) -> Self:
        """Reconstruct an instance from a database row.

        Returns ``Self`` so that ``Author.from_db(...)`` is typed as ``Author``,
        not as the base ``Model``.
        """
        ...

    # ── Inner Meta ─────────────────────────────────────────────────────
    class Meta:
        abstract:        ClassVar[bool]
        db_table:        ClassVar[str]
        ordering:        ClassVar[list[str]]
        unique_together: ClassVar[list[list[str]]]
        managed:         ClassVar[bool]
```

## How `@dataclass_transform` enables typed `__init__`

When pyright (or mypy with the plugin) encounters:

```python
class Author(Model):
    name  = CharField(max_length=100)         # inferred: CharField[str]
    email = CharField(max_length=200)          # inferred: CharField[str]
    bio   = TextField(null=True)               # inferred: TextField[str | None]
    age   = IntegerField(null=True)            # inferred: IntegerField[int | None]
```

It synthesises (for type-checking only, not at runtime):

```python
def __init__(
    self,
    *,
    name:  str,
    email: str,
    bio:   str | None,
    age:   int | None,
) -> None: ...
```

This synthesis uses the `__set__` signature of each field — `TextField[str | None].__set__`
takes `value: str | None`, so the `__init__` parameter for `bio` is `str | None`.

The `kw_only_default=True` in `@dataclass_transform` ensures all synthesised parameters are
keyword-only (matching how ORM models are typically constructed).

## How `Manager[Self]` resolves per-subclass

```python
class Model:
    objects: ClassVar[Manager[Self]]   # Self = the concrete subclass

class Author(Model): ...
class Article(Model): ...
```

Pyright resolves:
- `Author.objects` → `Manager[Author]`
- `Article.objects` → `Manager[Article]`
- `Author.objects.get(pk=1)` → `Coroutine[Any, Any, Author]`
- `Article.objects.filter(title__contains="ormpi")` → `QuerySet[Article]`
- `Author.objects.first()` → `Coroutine[Any, Any, Author | None]`

### Mypy compatibility note

Mypy's handling of `Self` inside `ClassVar` is tracked in
[mypy issue #14511](https://github.com/python/mypy/issues/14511). As of mypy 1.11,
`ClassVar[Manager[Self]]` is accepted but may emit spurious `[override]` warnings in deeply
nested hierarchies. If `mypy --strict` fails on `Manager[Self]`, use the following workaround:

```python
# mypy-friendly fallback — add this to mypy.ini overrides:
# [mypy-ormpi.models]
# disable_error_code = type-arg
```

Pyright (which the project treats as primary) handles `Manager[Self]` correctly in all cases.

## Verification Script

```python
# Run with: python -m pyright --pythonversion 3.12 python/ormpi/
# Or paste into a test file for reveal_type() checks.

import ormpi
from ormpi import Model, CharField, TextField, IntegerField, DateTimeField

class Author(Model):
    name       = CharField(max_length=100)
    email      = CharField(max_length=200, unique=True)
    bio        = TextField(null=True)
    age        = IntegerField(null=True)
    created_at = DateTimeField(auto_now_add=True)

# ── Field attribute types ──────────────────────────────────────────────────
async def check_field_types() -> None:
    author = await Author.objects.get(pk=1)

    reveal_type(author.name)        # Expected: str
    reveal_type(author.bio)         # Expected: str | None
    reveal_type(author.age)         # Expected: int | None
    reveal_type(author.created_at)  # Expected: datetime.datetime

    # Class access returns the descriptor
    reveal_type(Author.name)        # Expected: CharField[str]
    reveal_type(Author.bio)         # Expected: TextField[str | None]

# ── Manager / QuerySet types ───────────────────────────────────────────────
async def check_manager_types() -> None:
    reveal_type(Author.objects)                          # Expected: Manager[Author]
    reveal_type(Author.objects.filter(name="Alice"))     # Expected: QuerySet[Author]
    reveal_type(await Author.objects.get(pk=1))          # Expected: Author
    reveal_type(await Author.objects.create(name="x"))   # Expected: Author
    reveal_type(await Author.objects.first())            # Expected: Author | None
    reveal_type(await Author.objects.count())            # Expected: int

# ── Constructor type-checking ──────────────────────────────────────────────
def check_init_types() -> None:
    # OK — correct types
    Author(name="Alice", email="a@example.com", bio=None, age=30)

    # Error — name must be str
    Author(name=123)                   # type: ignore[arg-type]

    # Error — unknown field
    Author(unknown_field="x")          # type: ignore[unexpected-keyword-arg]

# ── from_db returns Self ───────────────────────────────────────────────────
def check_from_db() -> None:
    result = Author.from_db("default", ["name", "email"], ["Alice", "a@example.com"])
    reveal_type(result)   # Expected: Author (not Model)
```
