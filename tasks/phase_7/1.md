# Task 7.1 — ProjectState & MigrationAutoDetector

## Objective
Implement `ProjectState` (snapshot of all model definitions at a given migration version) and `MigrationAutoDetector` that diffs two states to generate migration operations.

## Location
`ormpi-core/src/migration/state.rs`
`ormpi-core/src/migration/detector.rs`

## ProjectState

```rust
// ormpi-core/src/migration/state.rs
use std::collections::HashMap;
use crate::fields::FieldDef;

/// A complete snapshot of the database schema at a given point.
#[derive(Debug, Clone, Default)]
pub struct ProjectState {
    pub models: HashMap<String, ModelState>,  // key: "app_label.ModelName"
}

#[derive(Debug, Clone)]
pub struct ModelState {
    pub app_label: String,
    pub name: String,
    pub fields: IndexMap<String, FieldState>,
    pub options: ModelOptions,
}

#[derive(Debug, Clone)]
pub struct FieldState {
    pub name: String,
    pub field_type_str: String,       // serializable string like "CharField(max_length=100)"
    pub field_def: FieldDef,
}

#[derive(Debug, Clone, Default)]
pub struct ModelOptions {
    pub db_table: Option<String>,
    pub ordering: Vec<String>,
    pub unique_together: Vec<Vec<String>>,
    pub indexes: Vec<IndexDef>,
    pub constraints: Vec<ConstraintDef>,
    pub managed: bool,
}

impl ProjectState {
    /// Build from the current ModelRegistry (represents current code state).
    pub fn from_registry() -> Self {
        let registry = ModelRegistry::global().read().unwrap();
        let mut state = ProjectState::default();
        for model_def in registry.all() {
            if model_def.abstract_model || model_def.proxy { continue; }
            let model_state = ModelState::from_model_def(model_def);
            let key = format!("{}.{}", model_def.app_label, model_def.model_name);
            state.models.insert(key, model_state);
        }
        state
    }

    /// Build by replaying migration files from disk.
    pub fn from_migrations(migration_dir: &Path) -> crate::Result<Self> {
        let mut state = ProjectState::default();
        let graph = MigrationGraph::load_from_dir(migration_dir)?;
        for migration_name in graph.topological_order() {
            let migration = graph.get(migration_name).unwrap();
            for op in &migration.operations {
                op.state_forwards(&mut state);
            }
        }
        Ok(state)
    }

    /// Apply a single operation to produce next state.
    pub fn apply_operation(&mut self, op: &dyn MigrationOperation) {
        op.state_forwards(self);
    }
}
```

## MigrationAutoDetector

```rust
// ormpi-core/src/migration/detector.rs

pub struct AutoDetector {
    pub from_state: ProjectState,   // current migration history state
    pub to_state: ProjectState,     // current code/registry state
}

impl AutoDetector {
    pub fn detect_changes(&self) -> Vec<Box<dyn MigrationOperation>> {
        let mut ops: Vec<Box<dyn MigrationOperation>> = Vec::new();

        // Step 1: Detect new models
        for (key, to_model) in &self.to_state.models {
            if !self.from_state.models.contains_key(key) {
                ops.push(Box::new(CreateModelOp {
                    app_label: to_model.app_label.clone(),
                    name: to_model.name.clone(),
                    fields: to_model.fields.values().map(|f| f.field_def.clone()).collect(),
                    options: to_model.options.clone(),
                }));
            }
        }

        // Step 2: Detect deleted models
        for (key, from_model) in &self.from_state.models {
            if !self.to_state.models.contains_key(key) {
                ops.push(Box::new(DeleteModelOp {
                    app_label: from_model.app_label.clone(),
                    name: from_model.name.clone(),
                }));
            }
        }

        // Step 3: Field-level changes
        for (key, to_model) in &self.to_state.models {
            if let Some(from_model) = self.from_state.models.get(key) {
                self.detect_field_changes(from_model, to_model, &mut ops);
            }
        }

        // Step 4: Order operations (CreateModel before AddField on same model, etc.)
        self.sort_operations(&mut ops);

        ops
    }

    fn detect_field_changes(
        &self,
        from: &ModelState,
        to: &ModelState,
        ops: &mut Vec<Box<dyn MigrationOperation>>,
    ) {
        // New fields
        for (fname, to_field) in &to.fields {
            if !from.fields.contains_key(fname) {
                ops.push(Box::new(AddFieldOp {
                    model: to.name.clone(),
                    name: fname.clone(),
                    field: to_field.field_def.clone(),
                }));
            }
        }

        // Removed fields
        for fname in from.fields.keys() {
            if !to.fields.contains_key(fname) {
                ops.push(Box::new(RemoveFieldOp {
                    model: to.name.clone(),
                    name: fname.clone(),
                }));
            }
        }

        // Changed fields
        for (fname, to_field) in &to.fields {
            if let Some(from_field) = from.fields.get(fname) {
                if from_field.field_type_str != to_field.field_type_str {
                    ops.push(Box::new(AlterFieldOp {
                        model: to.name.clone(),
                        name: fname.clone(),
                        old_field: from_field.field_def.clone(),
                        new_field: to_field.field_def.clone(),
                    }));
                }
            }
        }

        // Changed options (ordering, unique_together, indexes)
        if from.options != to.options {
            ops.push(Box::new(AlterModelOptionsOp {
                model: to.name.clone(),
                options: to.options.clone(),
            }));
        }
    }

    fn sort_operations(&self, ops: &mut Vec<Box<dyn MigrationOperation>>) {
        // Sort order priority:
        // 1. CreateModel (before any AddField on same model)
        // 2. AddField / AlterField
        // 3. RemoveField
        // 4. DeleteModel (after all field removals)
        ops.sort_by_key(|op| match op.operation_type() {
            OpType::CreateModel => 0,
            OpType::AddField    => 1,
            OpType::AlterField  => 2,
            OpType::AlterIndex  => 3,
            OpType::RemoveField => 4,
            OpType::DeleteModel => 5,
        });
    }
}
```

## Migration Name Generation

```rust
pub fn generate_migration_name(
    existing_migrations: &[String],
    ops: &[Box<dyn MigrationOperation>],
) -> String {
    let next_number = existing_migrations.len() + 1;
    let prefix = format!("{:04}", next_number);

    // Pick a descriptive suffix from the first operation
    let suffix = match ops.first() {
        Some(op) => match op.operation_type() {
            OpType::CreateModel => format!("initial"),
            OpType::AddField    => format!("add_{}", op.primary_identifier()),
            OpType::RemoveField => format!("remove_{}", op.primary_identifier()),
            OpType::AlterField  => format!("alter_{}", op.primary_identifier()),
            _                   => format!("auto"),
        },
        None => "auto".to_string(),
    };

    format!("{}_{}", prefix, suffix)
}
```

- [ ] No spurious changes when `from_state == to_state`
