# Task 2.1 — FieldDef Core Struct & FieldType Enum

## Objective
Define the central `FieldDef` struct and `FieldType` enum in Rust that all field types map to. This is the canonical representation of a database column used by the model registry, query builder, and migration engine.

## Location
`ormpi-core/src/fields/mod.rs`
`ormpi-core/src/fields/types.rs`

```rust
pub enum FieldType {
    // Auto primary keys
    AutoInt,
    AutoBigInt,

    Int,

    PositiveInt,

    Url { max_length: u32 },

    GenericIpAddress,

    Boolean,

    Date,
    Time,
    DateTime,
    Duration,

    // Binary / special
    Binary,
    Uuid,
    Json,
    JsonB,     // Postgres-specific

    // Relational (resolved separately — don't map to a single column)

        db_column: Option<String>,
    },
    OneToOne {

        on_delete: OnDelete,
    },

        to: String,
        through: Option<String>,  // intermediate model
```

## FieldDef Struct

```rust
// ormpi-core/src/fields/mod.rs
use std::fmt;

#[derive(Debug, Clone)]
pub struct FieldDef {
    pub name: String,
    pub attname: String,          // column name — usually same as name, or "{name}_id" for FK
    pub db_column: Option<String>,// explicit override
    pub field_type: FieldType,
    pub null: bool,
    pub blank: bool,
    pub default: DefaultValue,
    pub unique: bool,
    pub db_index: bool,
    pub primary_key: bool,
    pub editable: bool,
    pub choices: Vec<(OrmpiValue, String)>,  // (db_value, display_label)
    pub auto_now: bool,          // for DateTimeField
    pub auto_now_add: bool,      // for DateTimeField
    pub max_length: Option<u32>,
    pub validators: Vec<BuiltinValidator>,
    pub help_text: String,
    pub verbose_name: Option<String>,
    pub related_name: Option<String>,  // for relational fields
}

#[derive(Debug, Clone, PartialEq)]
pub enum DefaultValue {
    None,             // no default (required field unless null=True)
    Null,             // DEFAULT NULL
    Value(OrmpiValue),
    CurrentTimestamp, // auto_now_add=True equivalent
    EmptyString,
}

#[derive(Debug, Clone)]
pub enum BuiltinValidator {
    MinValue(f64),
    MaxValue(f64),
    MinLength(usize),
    MaxLength(usize),
    Regex(String),
    EmailValidator,
    UrlValidator,
    IpValidator,
}
```

## FieldDef → sea-query ColumnDef

```rust
impl FieldDef {
    /// Convert this field definition to a sea-query ColumnDef for DDL.
    /// Returns None for M2M fields (they don't produce a column in this table).
    pub fn to_column_def(&self) -> Option<sea_query::ColumnDef> {
        let col_name = self.column_name();
        let mut col = sea_query::ColumnDef::new(sea_query::Alias::new(&col_name));

        // Apply type
        match &self.field_type {
            FieldType::AutoInt => {
                col.integer().auto_increment().primary_key();
                return Some(col);
            }
            FieldType::AutoBigInt => {
                col.big_integer().auto_increment().primary_key();
                return Some(col);
            }
            FieldType::Int      => { col.integer(); }
            FieldType::BigInt   => { col.big_integer(); }
            FieldType::SmallInt => { col.small_integer(); }
            FieldType::Float    => { col.float(); }
            FieldType::Double   => { col.double(); }
            FieldType::Decimal { max_digits, decimal_places } => {
                col.decimal_len(*max_digits, *decimal_places);
            }
            FieldType::Char { max_length } => { col.string_len(*max_length); }
            FieldType::Text     => { col.text(); }
            FieldType::Boolean  => { col.boolean(); }
            FieldType::Date     => { col.date(); }
            FieldType::Time     => { col.time(); }
            FieldType::DateTime => { col.timestamp_with_time_zone(); }
            FieldType::Uuid     => { col.uuid(); }
            FieldType::Json     => { col.json(); }
            FieldType::JsonB    => { col.json_binary(); }
            FieldType::Binary   => { col.blob(); }
            FieldType::ForeignKey { .. } | FieldType::OneToOne { .. } => {
                col.big_integer(); // FK column stores the related PK
            }
            FieldType::ManyToMany { .. } => return None,  // no column
            // ... other types
        }

        // Apply constraints
        if self.primary_key { col.primary_key(); }
        if !self.null       { col.not_null(); }
        if self.unique      { col.unique_key(); }

        // Default value
        match &self.default {
            DefaultValue::None | DefaultValue::Null => {
                if self.null { col.default(sea_query::Value::from(None::<i32>)); }
            }
            DefaultValue::Value(v) => { col.default(v.to_sea_value()); }
            DefaultValue::CurrentTimestamp => { col.default(sea_query::Keyword::CurrentTimestamp); }
            DefaultValue::EmptyString => { col.default(""); }
        }

        Some(col)
    }

    /// The actual database column name (db_column override or attname).
    pub fn column_name(&self) -> String {
        if let Some(ref explicit) = self.db_column {
            explicit.clone()
        } else {
            self.attname.clone()
        }
    }
}
```

## Unit Tests

```rust
#[cfg(test)]
mod tests {
    use super::*;
    use sea_query::{MysqlQueryBuilder, PostgresQueryBuilder, SqliteQueryBuilder};

    fn make_field(ft: FieldType) -> FieldDef {
        FieldDef {
            name: "test".into(),
            attname: "test".into(),
            db_column: None,
            field_type: ft,
            null: false,
            blank: false,
            default: DefaultValue::None,
            unique: false,
            db_index: false,
            primary_key: false,
            editable: true,
            choices: vec![],
            auto_now: false,
            auto_now_add: false,
            max_length: None,
            validators: vec![],
            help_text: String::new(),
            verbose_name: None,
            related_name: None,
        }
    }

    #[test]
    fn test_integer_not_null() {
        let f = make_field(FieldType::Int);
        let col = f.to_column_def().unwrap();
        let sql = sea_query::Table::create()
            .table(sea_query::Alias::new("t"))
            .col(col)
            .to_string(SqliteQueryBuilder);
        assert!(sql.contains("NOT NULL"));
        assert!(sql.contains("integer"));
    }

    #[test]
    fn test_varchar_with_length() {
        let f = make_field(FieldType::Char { max_length: 100 });
        // verify VARCHAR(100) in MySQL
    }

    #[test]
    fn test_m2m_returns_none() {
        let f = make_field(FieldType::ManyToMany {
            to: "Other".into(), through: None, db_table: None,
        });
        assert!(f.to_column_def().is_none());
    }
}
```
