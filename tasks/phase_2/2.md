# Task 2.2 — Scalar & Temporal Field PyO3 Classes

## Objective
Implement all scalar, temporal, and special-purpose field classes as PyO3 `#[pyclass]` structs. Each class wraps a `FieldDef` and exposes the correct constructor kwargs matching Django's API. Fields are descriptors — define `__get__`/`__set__` so model instance attribute access works correctly.

## Base Field PyO3 Class

```rust
// ormpi-py/src/fields/mod.rs
use pyo3::prelude::*;
use ormpi_core::fields::{FieldDef, FieldType, DefaultValue};

/// All field classes share common constructor kwargs.
/// This macro generates the shared logic.
macro_rules! impl_field_class {
    ($cls:ident, $field_type_expr:expr) => {
        #[pymethods]
        impl $cls {
            #[new]
            #[pyo3(signature = (
                *,
                null = false,
                blank = false,
                default = None,
                unique = false,
                db_index = false,
                primary_key = false,
                editable = true,
                choices = None,
                db_column = None,
                help_text = "",
                verbose_name = None,
            ))]
            fn new(
                null: bool, blank: bool, default: Option<PyObject>,
                unique: bool, db_index: bool, primary_key: bool,
                editable: bool, choices: Option<Vec<(PyObject, String)>>,
                db_column: Option<String>, help_text: &str,
                verbose_name: Option<String>,
            ) -> PyResult<Self> {
                // Build FieldDef from kwargs
                Ok(Self {
                    inner: FieldDef {
                        name: String::new(),      // set by contribute_to_class
                        attname: String::new(),   // set by contribute_to_class
                        db_column,
                        field_type: $field_type_expr,
                        null, blank, unique, db_index, primary_key, editable,
                        default: parse_default(default)?,
                        choices: vec![],
                        auto_now: false, auto_now_add: false, max_length: None,
                        validators: vec![],
                        help_text: help_text.to_string(),
                        verbose_name, related_name: None,
                    }
                })
            }

            fn contribute_to_class(&mut self, _cls: &PyAny, name: String) -> PyResult<()> {
                self.inner.name = name.clone();
                self.inner.attname = name;
                Ok(())
            }

            // Descriptor protocol
            fn __get__(&self, py: Python<'_>, obj: Option<&PyAny>, _objtype: Option<&PyAny>)
                -> PyResult<PyObject>
            {
                match obj {
                    None => Ok(py.None()), // accessed on class → return field itself
                    Some(instance) => {
                        // Read from instance._ormpi_values dict
                        let values = instance.getattr("_ormpi_values")?;
                        values.get_item(&self.inner.name)
                            .unwrap_or_else(|_| Ok(py.None()))
                    }
                }
            }

            fn __set__(&self, instance: &PyAny, value: PyObject) -> PyResult<()> {
                let values = instance.getattr("_ormpi_values")?;
                values.set_item(&self.inner.name, value)?;
                // Mark field as dirty
                let dirty = instance.getattr("_ormpi_dirty")?;
                dirty.call_method1("add", (&self.inner.name,))?;
                Ok(())
            }

            #[getter]
            fn name(&self) -> &str { &self.inner.name }

            #[getter]
            fn null(&self) -> bool { self.inner.null }

            #[getter]
            fn unique(&self) -> bool { self.inner.unique }

            #[getter]
            fn primary_key(&self) -> bool { self.inner.primary_key }

            fn __repr__(&self) -> String {
                format!("<{} '{}'>", stringify!($cls), self.inner.name)
            }
        }
    };
}
```

// DateTimeField (has auto_now and auto_now_add)
#[pyclass(name = "DateTimeField")]
pub struct PyDateTimeField { inner: FieldDef }

#[pymethods]
impl PyDateTimeField {
    #[new]
    #[pyo3(signature = (*, auto_now=false, auto_now_add=false, null=false, ...))]
    fn new(auto_now: bool, auto_now_add: bool, null: bool, ...) -> PyResult<Self> {
        if auto_now && auto_now_add {
            return Err(PyValueError::new_err("auto_now and auto_now_add are mutually exclusive"));
        }
        let default = if auto_now_add {
            DefaultValue::CurrentTimestamp
        } else {
            DefaultValue::None
        };
        Ok(Self { inner: FieldDef {
            field_type: FieldType::DateTime,
            auto_now, auto_now_add,
            default, null: null || auto_now || auto_now_add,
            ..Default::default()
        }})
    }
}
