# Task 2.3 — Relational Field Classes (ForeignKey, OneToOne, ManyToMany)

## Objective
Implement the three relational field PyO3 classes. These differ fundamentally from scalar fields: they reference other models, produce extra columns (`_id` suffix for FK/O2O), generate FK constraints in DDL, and create junction tables for M2M relationships.

## Location
`ormpi-py/src/fields/relational.rs`

## ForeignKey

```rust
#[pyclass(name = "ForeignKey")]
pub struct PyForeignKey {
    inner: FieldDef,
    pub to: String,               // "ModelName" or "app.ModelName" or "self"
    pub on_delete: OnDelete,
    pub related_name: Option<String>,
    pub related_query_name: Option<String>,
    pub to_field: Option<String>, // which field on `to` to reference (default: pk)
    pub limit_choices_to: Option<PyObject>,
}

#[pymethods]
impl PyForeignKey {
    #[new]
    #[pyo3(signature = (
        to,
        on_delete,
        *,
        related_name = None,
        related_query_name = None,
        to_field = None,
        db_constraint = true,
        null = false,
        blank = false,
        db_index = true,         // FK fields are indexed by default
        db_column = None,
    ))]
    fn new(
        py: Python<'_>,
        to: PyObject,            // can be str or class
        on_delete: &str,
        related_name: Option<String>,
        related_query_name: Option<String>,
        to_field: Option<String>,
        db_constraint: bool,
        null: bool, blank: bool,
        db_index: bool,
        db_column: Option<String>,
    ) -> PyResult<Self> {
        let to_str = resolve_to_string(py, &to)?;
        let on_delete = parse_on_delete(on_delete)?;

        Ok(Self {
            inner: FieldDef {
                name: String::new(),
                attname: String::new(),  // set to "{name}_id" in contribute_to_class
                db_column,
                field_type: FieldType::ForeignKey {
                    to: to_str.clone(),
                    on_delete,
                    db_column: None,
                },
                null, blank,
                db_index,
                unique: false,
                primary_key: false,
                editable: true,
                default: DefaultValue::None,
                choices: vec![],
                auto_now: false, auto_now_add: false, max_length: None,
                validators: vec![],
                help_text: String::new(),
                verbose_name: None,
                related_name: related_name.clone(),
            },
            to: to_str,
            on_delete,
            related_name,
            related_query_name,
            to_field,
            limit_choices_to: None,
        })
    }

    fn contribute_to_class(&mut self, cls: &PyAny, name: String) -> PyResult<()> {
        self.inner.name = name.clone();
        self.inner.attname = format!("{}_id", name);  // FK stores {name}_id
        // Register reverse relation on the target model (done at model registration)
        Ok(())
    }

    // Forward access: article.author → fetches Author by author_id
    fn __get__(&self, py: Python<'_>, obj: Option<&PyAny>, _objtype: Option<&PyAny>)
        -> PyResult<PyObject>
    {
        match obj {
            None => Ok(py.None()),  // accessed on class → return field descriptor
            Some(instance) => {
                // Check if related object is cached
                let cache_key = format!("_ormpi_cache_{}", self.inner.name);
                if let Ok(cached) = instance.getattr(cache_key.as_str()) {
                    if !cached.is_none() { return Ok(cached.into()); }
                }
                // Return a deferred accessor (lazy load)
                // The actual fetch happens when the coroutine is awaited
                let values = instance.getattr("_ormpi_values")?;
                let id_val = values.get_item(&self.inner.attname)?;
                if id_val.is_none() { return Ok(py.None()); }
                // Return a RelatedObjectDescriptor that can be awaited
                Ok(py.None())  // placeholder — full impl in Phase 4
            }
        }
    }

    fn __set__(&self, instance: &PyAny, value: PyObject) -> PyResult<()> {
        // Setting article.author = author_instance: store id
        let py = value.py();
        if value.is_none(py) {
            let values = instance.getattr("_ormpi_values")?;
            values.set_item(&self.inner.attname, py.None())?;
        } else {
            let pk = value.getattr("pk")?;
            let values = instance.getattr("_ormpi_values")?;
            values.set_item(&self.inner.attname, pk)?;
            // Cache the object
            instance.setattr(format!("_ormpi_cache_{}", self.inner.name).as_str(), value)?;
        }
        Ok(())
    }

    #[getter]
    fn to_model(&self) -> &str { &self.to }

    #[getter]
    fn attname(&self) -> String { format!("{}_id", self.inner.name) }

    #[getter]
    fn column(&self) -> String {
        self.inner.db_column.clone().unwrap_or_else(|| format!("{}_id", self.inner.name))
    }
}
```

pub struct PyManyToManyField {
    inner: FieldDef,
    pub to: String,
    pub through: Option<String>,   // explicit intermediate model
    pub db_table: Option<String>,  // junction table name override
    pub related_name: Option<String>,
    pub symmetrical: bool,         // for self-referential M2M
}

#[pymethods]
impl PyManyToManyField {
    #[new]
    #[pyo3(signature = (to, *, through=None, db_table=None, related_name=None, symmetrical=None))]
    fn new(
        py: Python<'_>,
        to: PyObject,
        through: Option<PyObject>,
        db_table: Option<String>,
        related_name: Option<String>,
        symmetrical: Option<bool>,
    ) -> PyResult<Self> {
        let to_str = resolve_to_string(py, &to)?;
        let through_str = through.map(|t| resolve_to_string(py, &t)).transpose()?;
        let sym = symmetrical.unwrap_or_else(|| to_str == "self");

        Ok(Self {
            inner: FieldDef {
                name: String::new(),
                attname: String::new(),
                field_type: FieldType::ManyToMany {
                    to: to_str.clone(),
                    through: through_str.clone(),
                    db_table: db_table.clone(),
                },
                ..Default::default()
            },
            to: to_str,
            through: through_str,
            db_table,
            related_name,
            symmetrical: sym,
        })
    }

    fn contribute_to_class(&mut self, _cls: &PyAny, name: String) -> PyResult<()> {
        self.inner.name = name;

## Helper Functions

```rust
fn resolve_to_string(py: Python<'_>, to: &PyObject) -> PyResult<String> {
    if let Ok(s) = to.extract::<String>(py) {
        return Ok(s);  // string reference like "Author"
    }
    // Extract class name from class object
    if let Ok(cls) = to.extract::<&PyType>(py) {
        return Ok(cls.name()?.to_string());
    }
    Err(PyTypeError::new_err("'to' must be a Model class or string"))
}

fn parse_on_delete(s: &str) -> PyResult<OnDelete> {
    match s.to_uppercase().as_str() {
        "CASCADE"    => Ok(OnDelete::Cascade),
        "PROTECT"    => Ok(OnDelete::Protect),
        "SET_NULL"   => Ok(OnDelete::SetNull),
        "SET_DEFAULT"=> Ok(OnDelete::SetDefault),
        "DO_NOTHING" => Ok(OnDelete::DoNothing),
        "RESTRICT"   => Ok(OnDelete::Restrict),
        _            => Err(PyValueError::new_err(format!("Unknown on_delete: {}", s))),
    }
}
```
