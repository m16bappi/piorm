## Q Object (Rust)

```rust
// ormpi-core/src/query/q.rs
#[derive(Debug, Clone)]
pub struct Q {
    pub connector: QConnector,
    pub negated: bool,
    pub children: Vec<QChild>,
}

#[derive(Debug, Clone, Copy)]
pub enum QConnector { And, Or }

#[derive(Debug, Clone)]
pub enum QChild {
    Leaf { key: String, value: OrmpiValue },
    Branch(Box<Q>),
}

impl Q {
    pub fn and(self, other: Q) -> Q {
        Q { connector: QConnector::And, negated: false, children: vec![
            QChild::Branch(Box::new(self)),
            QChild::Branch(Box::new(other)),
        ]}
    }

    pub fn or(self, other: Q) -> Q {
        Q { connector: QConnector::Or, negated: false, children: vec![
            QChild::Branch(Box::new(self)),
            QChild::Branch(Box::new(other)),
        ]}
    }

    pub fn negate(self) -> Q {
        Q { negated: !self.negated, ..self }
    }
}
```

## PyQ Class (Python-facing)

```rust
#[pyclass(name = "Q")]
pub struct PyQ { inner: Q }

#[pymethods]
impl PyQ {
    #[new]
    #[pyo3(signature = (**kwargs))]
    fn new(py: Python<'_>, kwargs: Option<&PyDict>) -> PyResult<Self> {
        let children = kwargs.map(|kw| {
            kw.iter().map(|(k, v)| {
                let key: String = k.extract()?;
                let val = py_to_ormpi_value(py, v)?;
                Ok(QChild::Leaf { key, value: val })
            }).collect::<PyResult<Vec<_>>>()
        }).transpose()?.unwrap_or_default();

        Ok(Self { inner: Q { connector: QConnector::And, negated: false, children } })
    }

    fn __and__(&self, other: &PyQ) -> PyQ {
        PyQ { inner: self.inner.clone().and(other.inner.clone()) }
    }

    fn __or__(&self, other: &PyQ) -> PyQ {
        PyQ { inner: self.inner.clone().or(other.inner.clone()) }
    }

    fn __invert__(&self) -> PyQ {
        PyQ { inner: self.inner.clone().negate() }
    }
}
```

## get()

```rust
fn get<'py>(slf: &'py PyAny, py: Python<'py>, args: &PyTuple, kwargs: Option<&PyDict>) -> PyResult<&'py PyAny> {
    // Apply filter from args/kwargs, set limit=2
    let qs = slf.call_method(py, "filter", args, kwargs)?;
    let mut state = get_state(qs.as_ref(py))?.clone();
    state.limit = Some(2);   // fetch 2 to detect MultipleObjectsReturned

    let model_name = state.model_name.clone();
    pyo3_async_runtimes::tokio::future_into_py(py, async move {
        let rows = evaluate_rows(&state).await?;
        match rows.len() {
            0 => Err(OrmpiError::DoesNotExist { model: model_name }),
            1 => Ok(rows.into_iter().next().unwrap()),
            _ => Err(OrmpiError::MultipleObjectsReturned { model: model_name, count: rows.len() }),
        }
    })
}
```

## create()

```rust
fn create<'py>(slf: &'py PyAny, py: Python<'py>, kwargs: Option<&PyDict>) -> PyResult<&'py PyAny> {
    let state = get_state(slf)?;
    let using = state.database.clone().unwrap_or("default".to_string());
    let model_name = state.model_name.clone();

    pyo3_async_runtimes::tokio::future_into_py(py, async move {
        Python::with_gil(|py| {
            // 1. Instantiate the model
            let model_cls = get_model_class(py, &model_name)?;
            let instance = model_cls.call(PyTuple::empty(py), kwargs)?;
            // 2. Call save()
            let coro = instance.call_method("save", (using,), None)?;
            Ok(coro.into())
        })
    })
}
```

## Bulk update() / delete()

```rust
fn update<'py>(slf: &'py PyAny, py: Python<'py>, kwargs: Option<&PyDict>) -> PyResult<&'py PyAny> {
    let state = get_state(slf)?.clone();
    let updates: HashMap<String, OrmpiValue> = parse_kwargs_to_values(py, kwargs)?;

    pyo3_async_runtimes::tokio::future_into_py(py, async move {
        let pool = get_pool(&state)?;
        let backend = QueryBackend(pool.backend());
        let model_def = get_model_def(&state)?;

        let table = Alias::new(&model_def.table_name);
        let mut stmt = Query::update().table(table).to_owned();

        for (field_name, value) in &updates {
            let col = resolve_field_to_column(field_name, model_def)?;
            stmt.value(Alias::new(&col), value.to_sea_value());
        }

        // Apply WHERE from state filters
        apply_conditions_to_update(&mut stmt, &state.filters, model_def);

        let (sql, values) = backend.build_update(&stmt);
        let rows_affected = execute_affecting(pool, &sql, &values).await?;
        Ok(rows_affected as i64)
    })
}

fn delete<'py>(slf: &'py PyAny, py: Python<'py>) -> PyResult<&'py PyAny> {
    let state = get_state(slf)?.clone();
    pyo3_async_runtimes::tokio::future_into_py(py, async move {
        let pool = get_pool(&state)?;
        let backend = QueryBackend(pool.backend());
        let model_def = get_model_def(&state)?;

        let table = Alias::new(&model_def.table_name);
        let mut stmt = Query::delete().from_table(table).to_owned();
        apply_conditions_to_delete(&mut stmt, &state.filters, model_def);

        let (sql, values) = backend.build_delete(&stmt);
        let count = execute_affecting(pool, &sql, &values).await?;
        let mut breakdown = HashMap::new();
        breakdown.insert(model_def.model_name.clone(), count);
        Ok((count, breakdown))
    })
}
```

## Aggregation Functions

```rust
// ormpi-core/src/query/aggregate.rs
#[derive(Debug, Clone)]
pub enum AggregateFunc {
    Count { field: Option<String>, distinct: bool },
    Sum { field: String },
    Avg { field: String },
    Max { field: String },
    Min { field: String },
    StdDev { field: String, sample: bool },
    Variance { field: String, sample: bool },
}

impl AggregateFunc {
    pub fn to_sea_expr(&self) -> sea_query::Expr {
        match self {
            Self::Count { field: None, .. }  => Func::count(Expr::asterisk()),
            Self::Count { field: Some(f), distinct } => {
                if *distinct { Func::count_distinct(Alias::new(f)) }
                else { Func::count(Alias::new(f)) }
            }
            Self::Sum { field }  => Func::sum(Alias::new(field)),
            Self::Avg { field }  => Func::avg(Alias::new(field)),
            Self::Max { field }  => Func::max(Alias::new(field)),
            Self::Min { field }  => Func::min(Alias::new(field)),
        }
    }
}
```

```python
# Exported to Python:
from ormpi import Count, Sum, Avg, Max, Min

# Usage:
result = await Article.objects.aggregate(
    total=Count("id"),
    avg_views=Avg("views"),
)
# result = {"total": 42, "avg_views": 150.3}

# annotate() adds per-row annotation:
qs = Article.objects.annotate(comment_count=Count("comments"))
articles = await qs.filter(comment_count__gt=5)
```

## Acceptance Criteria
- [x] `Q(title="foo") & Q(title="bar")` creates AND condition
- [x] `Q(a=1) | Q(b=2)` creates OR condition
- [x] `~Q(active=True)` creates NOT condition
- [x] `await Article.objects.get(id=1)` returns single instance
- [x] `.get()` raises `DoesNotExist` when not found
- [x] `.get()` raises `MultipleObjectsReturned` when >1 result
- [x] `await Article.objects.create(title="x")` creates and returns instance
- [x] `await Article.objects.filter(...).update(title="new")` returns affected row count
