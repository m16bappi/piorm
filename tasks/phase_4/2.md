## State → SelectStatement

```rust
// ormpi-core/src/query/evaluate.rs

pub fn build_select(state: &QueryState, model_def: &ModelDef) -> Result<SelectStatement> {
    let mut stmt = Query::select();
    let table = Alias::new(&model_def.table_name);

    // SELECT columns
    match &state.select_fields {
        SelectFields::All => { stmt.from(table.clone()).expr(Expr::asterisk()); }
        SelectFields::Only(fields) => {
            stmt.from(table.clone());
            for f in fields {
                let col = Alias::new(f);
                stmt.column(col);
            }
        }
        SelectFields::Defer(deferred) => {
            stmt.from(table.clone());
            for field in model_def.fields.values() {
                if !deferred.contains(&field.attname) {
                    stmt.column(Alias::new(&field.column_name()));
                }
            }
        }
    }

    // Annotations (added as extra columns)
    for (alias, annotation) in &state.annotations {
        stmt.expr_as(annotation.to_sea_expr(), Alias::new(alias));
    }

    // WHERE (filters)
    for filter_node in &state.filters.filters {
        let cond = filter_node_to_expr(filter_node, model_def, false);
        stmt.cond_where(cond);
    }

    // WHERE NOT (excludes)
    for filter_node in &state.filters.excludes {
        let cond = filter_node_to_expr(filter_node, model_def, true);
        stmt.cond_where(cond);
    }

    // ORDER BY
    for order in &state.order_by {
        let col = Alias::new(&order.field);
        let direction = if order.descending { Order::Desc } else { Order::Asc };
        stmt.order_by(col, direction);
    }

    // Default ordering from Meta
    if state.order_by.is_empty() {
        for order in &model_def.ordering {
            let col = Alias::new(&order.field);
            let direction = if order.descending { Order::Desc } else { Order::Asc };
            stmt.order_by(col, direction);
        }
    }

    // DISTINCT
    if state.distinct { stmt.distinct(); }

    // LIMIT / OFFSET
    if let Some(limit) = state.limit { stmt.limit(limit); }
    if let Some(offset) = state.offset { stmt.offset(offset); }

    Ok(stmt.to_owned())
}
```

## Execute & Deserialize

```rust
pub async fn evaluate(
    state: &QueryState,
    pool: &OrmpiPool,
    backend: QueryBackend,
    model_def: &ModelDef,
) -> Result<Vec<OrmpiRow>> {
    let stmt = build_select(state, model_def)?;
    let (sql, values) = backend.build_select(&stmt);

    let rows = match pool {
        OrmpiPool::Sqlite(p) => {
            let mut q = sqlx::query(&sql);
            for val in &values.0 { q = bind_sqlite(q, val); }
            q.fetch_all(p).await?
                .into_iter()
                .map(|r| sqlite_row_to_ormpi(&r))
                .collect::<Result<Vec<_>>>()?
        }
        OrmpiPool::Mysql(p)    => { /* similar */ }
        OrmpiPool::Postgres(p) => { /* similar */ }
    };

    Ok(rows)
}

/// Deserialize OrmpiRows into Python model instances
pub fn rows_to_model_instances(
    py: Python<'_>,
    rows: Vec<OrmpiRow>,
    model_def: &ModelDef,
    model_cls: &PyType,
) -> PyResult<Vec<PyObject>> {
    rows.into_iter().map(|row| {
        let instance = model_cls.call0()?;  // Model.__new__(cls)
        let values = PyDict::new(py);
        for (col_name, value) in &row.values {
            values.set_item(col_name, value_to_py(py, value))?;
        }
        instance.setattr("_ormpi_values", values)?;
        let state = PyDict::new(py);
        state.set_item("adding", false)?;
        state.set_item("db", py.None())?;
        instance.setattr("_ormpi_state", state)?;
        instance.setattr("_ormpi_dirty", PySet::empty(py)?)?;
        Ok(instance.into())
    }).collect()
}
```

## __getitem__ (slicing)

```rust
fn __getitem__(&self, py: Python<'_>, key: &PyAny) -> PyResult<PyObject> {
    if let Ok(idx) = key.extract::<i64>() {
        // Integer index → fetch single item
        let mut state = self.state.clone();
        state.limit = Some(1);
        state.offset = Some(if idx >= 0 { idx as u64 } else { 0 });
        // Return coroutine that fetches single item
        Ok(PyQuerySet { state }.into_py(py))
    } else if let Ok(slice) = key.downcast::<PySlice>() {
        // Slice → set LIMIT and OFFSET
        let (start, stop, _step) = slice.indices(i64::MAX)?;
        let mut state = self.state.clone();
        if start > 0 { state.offset = Some(start as u64); }
        if stop > 0 { state.limit = Some((stop - start) as u64); }
        Ok(PyQuerySet { state }.into_py(py))
    } else {
        Err(PyTypeError::new_err("QuerySet indices must be integers or slices"))
    }
}
```
