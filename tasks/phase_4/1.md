```rust
#[derive(Debug, Clone, PartialEq)]
pub enum Lookup {
    Exact,
    IExact,
    Contains,
    IContains,
    StartsWith,
    IStartsWith,
    EndsWith,
    IEndsWith,
    In,
    Gt, Gte, Lt, Lte,
    Range,
    IsNull,
    Regex,
    IRegex,
    Date, Year, Month, Day, Hour, Minute, Second,
    Week, WeekDay, Quarter,
}

impl Lookup {
    pub fn from_str(s: &str) -> Option<Self> {
        match s {
            "exact"       => Some(Self::Exact),
            "iexact"      => Some(Self::IExact),
            "contains"    => Some(Self::Contains),
            "icontains"   => Some(Self::IContains),
            "startswith"  => Some(Self::StartsWith),
            "istartswith" => Some(Self::IStartsWith),
            "endswith"    => Some(Self::EndsWith),
            "iendswith"   => Some(Self::IEndsWith),
            "in"          => Some(Self::In),
            "gt"          => Some(Self::Gt),
            "gte"         => Some(Self::Gte),
            "lt"          => Some(Self::Lt),
            "lte"         => Some(Self::Lte),
            "range"       => Some(Self::Range),
            "isnull"      => Some(Self::IsNull),
            "regex"       => Some(Self::Regex),
            "iregex"      => Some(Self::IRegex),
            "date"        => Some(Self::Date),
            "year"        => Some(Self::Year),
            "month"       => Some(Self::Month),
            "day"         => Some(Self::Day),
            _             => None,
        }
    }
}
```

## Kwarg Parsing: "field__related__lookup" → FilterNode

```rust
pub fn parse_filter_kwarg(
    key: &str,
    value: OrmpiValue,
    model_def: &ModelDef,
    registry: &ModelRegistry,
) -> Result<FilterNode> {
    let parts: Vec<&str> = key.split("__").collect();

    // Determine if last part is a known lookup
    let (field_parts, lookup) = if let Some(lk) = Lookup::from_str(parts.last().unwrap()) {
        (&parts[..parts.len()-1], lk)
    } else {
        (parts.as_slice(), Lookup::Exact)
    };

    // Validate field path exists in model (follow FK traversals)
    validate_field_path(field_parts, model_def, registry)?;

    Ok(FilterNode::Lookup {
        field_path: field_parts.iter().map(|s| s.to_string()).collect(),
        lookup,
        value,
    })
}
```

## FilterNode → sea-query Expr

```rust
pub fn filter_node_to_expr(
    node: &FilterNode,
    model_def: &ModelDef,
    negated: bool,
) -> sea_query::Condition {
    match node {
        FilterNode::Lookup { field_path, lookup, value } => {
            let col = resolve_column_ref(field_path, model_def);
            let expr = match lookup {
                Lookup::Exact     => Expr::col(col).eq(value.to_sea_value()),
                Lookup::IExact    => Expr::col(col).eq(Func::lower(value.to_sea_value())),
                Lookup::Contains  => Expr::col(col).like(format!("%{}%", value.as_str())),
                Lookup::IContains => Expr::col(col).ilike(format!("%{}%", value.as_str())),
                Lookup::In        => Expr::col(col).is_in(value.as_list_sea()),
                Lookup::IsNull    => {
                    if value.as_bool() { Expr::col(col).is_null() }
                    else { Expr::col(col).is_not_null() }
                },
                Lookup::Gt  => Expr::col(col).gt(value.to_sea_value()),
                Lookup::Gte => Expr::col(col).gte(value.to_sea_value()),
                Lookup::Lt  => Expr::col(col).lt(value.to_sea_value()),
                Lookup::Lte => Expr::col(col).lte(value.to_sea_value()),
                // ... other lookups
            };
            if negated { Condition::all().add(expr.not()) }
            else { Condition::all().add(expr) }
        }
        FilterNode::Q(q) => q_to_condition(q, negated),
    }
}
```

```rust
#[pyclass(name = "QuerySet")]
pub struct PyQuerySet {
    pub state: QueryState,
}

#[pymethods]
impl PyQuerySet {
    fn all(&self) -> Self {
        Self { state: self.state.clone() }
    }

    fn none(&self) -> Self {
        let mut state = self.state.clone();
        state.filters.filters.push(FilterNode::Lookup {
            field_path: vec!["pk".to_string()],
            lookup: Lookup::IsNull,
            value: OrmpiValue::Bool(true),  // pk IS NULL → always empty
        });
        Self { state }
    }

    fn filter(&self, py: Python<'_>, args: &PyTuple, kwargs: Option<&PyDict>) -> PyResult<Self> {
        let mut state = self.state.clone();
        // Parse Q objects from args
        for q_obj in args.iter() {
            let q = extract_q_node(py, q_obj)?;
            state.filters.filters.push(FilterNode::Q(q));
        }
        // Parse field=value kwargs
        if let Some(kw) = kwargs {
            let registry = ModelRegistry::global().read().unwrap();
            let model_def = registry.get(&state.app_label, &state.model_name)
                .ok_or_else(|| ...)?;
            for (key, value) in kw.iter() {
                let key: String = key.extract()?;
                let val = py_to_ormpi_value(py, value)?;
                let node = parse_filter_kwarg(&key, val, model_def, &registry)?;
                state.filters.filters.push(node);
            }
        }
        Ok(Self { state })
    }

        let mut state = self.state.clone();

        // ...
        Ok(Self { state })
    }

    fn order_by(&self, py: Python<'_>, fields: Vec<String>) -> PyResult<Self> {
        let mut state = self.state.clone();
        state.order_by = fields.iter().map(|f| parse_order_spec(f)).collect();
        Ok(Self { state })
    }

        let mut state = self.state.clone();
        state.distinct = true;
        Self { state }
    }

        let mut state = self.state.clone();
        state.database = Some(alias);
        Self { state }

        let mut state = self.state.clone();
        state.select_fields = SelectFields::Only(fields);
        Self { state }

        let mut state = self.state.clone();
        state.select_fields = SelectFields::Defer(fields);
        Self { state }
    }

    fn __repr__(&self) -> String {
        format!("<QuerySet [{}.{}]>", self.state.app_label, self.state.model_name)
    }
```
