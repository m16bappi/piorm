# Task 8.1 — CLI: makemigrations, migrate, showmigrations

## Objective
Implement the three primary CLI commands using `clap`: `makemigrations` (detect and write migration files), `migrate` (apply/reverse), and `showmigrations` (status display).

## CLI Root (clap)

```rust
// ormpi-cli/src/main.rs
use clap::{Parser, Subcommand};

#[derive(Parser)]
#[command(
    name = "ormpi",
    about = "OrmPi database management commands",
    version = env!("CARGO_PKG_VERSION"),
)]
struct Cli {
    #[arg(long, env = "ORMPI_SETTINGS_MODULE", global = true)]
    settings: Option<String>,

    #[arg(long, short = 'v', default_value = "1", global = true)]
    verbosity: u8,

    #[arg(long, global = true)]
    no_color: bool,

    #[command(subcommand)]
    command: Commands,
}

#[derive(Subcommand)]
enum Commands {
    /// Detect model changes and create migration files
    Makemigrations(MakemigrationsArgs),
    /// Apply or reverse database migrations
    Migrate(MigrateArgs),
    /// List all migrations and their status
    Showmigrations(ShowmigrationsArgs),
    /// Print SQL for a migration without running it
    Sqlmigrate(SqlmigrateArgs),
    /// Squash migrations into a single file
    Squashmigrations(SquashArgs),
    /// Open an interactive database shell
    Dbshell(DbshellArgs),
    /// Introspect database and output model definitions
    Inspectdb(InspectdbArgs),
    /// Print version info
    Version,
}

fn main() {
    let cli = Cli::parse();

    // Initialize Python interpreter
    pyo3::prepare_freethreaded_python();

    // Run the command
    let result = Python::with_gil(|py| {
        // Load settings
        let settings = load_settings(py, cli.settings.as_deref())?;
        // Initialize ormpi
        ormpi_core::runtime::block_on(ormpi_core::registry::initialize(settings.databases))?;

        match cli.command {
            Commands::Makemigrations(args) => run_makemigrations(py, args, cli.verbosity),
            Commands::Migrate(args)        => run_migrate(py, args, cli.verbosity),
            Commands::Showmigrations(args) => run_showmigrations(py, args),
            Commands::Sqlmigrate(args)     => run_sqlmigrate(py, args),
            Commands::Version              => { println!("ormpi {}", env!("CARGO_PKG_VERSION")); Ok(0) }
            // ...
        }
    });

    std::process::exit(result.unwrap_or(1));
}
```

## makemigrations Command

```rust
// ormpi-cli/src/commands/makemigrations.rs
#[derive(clap::Args)]
pub struct MakemigrationsArgs {
    /// App labels to detect changes for (default: all)
    pub apps: Vec<String>,

    #[arg(long)]
    pub name: Option<String>,

    #[arg(long, help = "Create an empty migration")]
    pub empty: bool,

    #[arg(long, help = "Show what would be created, don't write files")]
    pub dry_run: bool,

    #[arg(long, help = "Exit with non-zero if changes detected (CI)")]
    pub check: bool,
}

pub fn run_makemigrations(py: Python<'_>, args: MakemigrationsArgs, verbosity: u8) -> PyResult<i32> {
    let migration_dir = find_migrations_dir(py)?;

    // 1. Load current code state from ModelRegistry
    let to_state = ProjectState::from_registry();

    // 2. Load migration history state
    let from_state = ProjectState::from_migrations(&migration_dir)
        .unwrap_or_default();

    // 3. Detect changes
    let detector = AutoDetector { from_state, to_state };
    let ops = detector.detect_changes();

    if ops.is_empty() {
        if verbosity >= 1 { println!("No changes detected"); }
        return Ok(0);
    }

    if args.check {
        // For CI: exit 1 if there are pending changes
        eprintln!("Model changes detected but not yet migrated.");
        return Ok(1);
    }

    if args.dry_run {
        println!("Would create migration with {} operations:", ops.len());
        for op in &ops { println!("  - {}", op.describe()); }
        return Ok(0);
    }

    // 4. Generate migration name
    let existing = list_migration_files(&migration_dir)?;
    let name = args.name.unwrap_or_else(|| generate_migration_name(&existing, &ops));

    // 5. Find dependencies (last migration in each app)
    let deps = find_dependencies(&existing, &ops);

    // 6. Write migration file
    let migration_file = MigrationFile {
        migration: MigrationMeta {
            name: name.clone(),
            dependencies: deps,
            replaces: vec![],
        },
        operations: ops.iter().map(|op| op.to_record()).collect(),
    };

    let file_path = migration_dir.join(format!("{}.toml", name));
    migration_file.write_to_file(&file_path)?;

    if verbosity >= 1 {
        println!("Migrations for '{}':", find_app_label(&migration_dir));
        println!("  {}", file_path.display());
        for op in &ops { println!("    - {}", op.describe()); }
    }

    Ok(0)
}
```

## migrate Command

```rust
#[derive(clap::Args)]
pub struct MigrateArgs {
    pub app_label: Option<String>,
    pub migration_name: Option<String>,   // "zero" for full rollback

    #[arg(long, help = "Mark migrations as applied without running SQL")]
    pub fake: bool,

    #[arg(long, help = "Show plan without executing")]
    pub plan: bool,

    #[arg(long)]
    pub database: Option<String>,
}

pub fn run_migrate(py: Python<'_>, args: MigrateArgs, verbosity: u8) -> PyResult<i32> {
    let migration_dir = find_migrations_dir(py)?;
    let db_alias = args.database.as_deref().unwrap_or("default");

    let registry = DatabaseRegistry::global().read().unwrap();
    let pool = registry.get_pool(db_alias)?;
    let backend = QueryBackend(pool.backend());

    let recorder = MigrationRecorder::new(pool.clone());
    let graph = MigrationGraph::load_from_dir(&migration_dir)?;

    // Ensure ormpi_migrations table exists
    ormpi_core::runtime::block_on(recorder.ensure_schema())?;

    let executor = MigrationExecutor { recorder, graph, pool: pool.clone(), backend };

    if args.plan {
        let applied = ormpi_core::runtime::block_on(executor.recorder.applied())?;
        let pending = executor.graph.migrations_to_apply(None, &applied);
        for name in pending { println!("  [ ] {}", name); }
        return Ok(0);
    }

    let target = args.migration_name.as_deref();

    if target == Some("zero") {
        // Full rollback
        ormpi_core::runtime::block_on(executor.rollback_to("", args.fake))?;
    } else {
        ormpi_core::runtime::block_on(executor.migrate(target, args.fake))?;
    }

    Ok(0)
}
```

## showmigrations Command

```rust
#[derive(clap::Args)]
pub struct ShowmigrationsArgs {
    pub apps: Vec<String>,
    #[arg(long, short = 'p')]
    pub plan: bool,
    #[arg(long)]
    pub database: Option<String>,
}

pub fn run_showmigrations(py: Python<'_>, args: ShowmigrationsArgs) -> PyResult<i32> {
    let migration_dir = find_migrations_dir(py)?;
    let db_alias = args.database.as_deref().unwrap_or("default");

    let pool = get_pool(db_alias)?;
    let recorder = MigrationRecorder::new(pool.clone());
    let graph = MigrationGraph::load_from_dir(&migration_dir)?;

    let applied = ormpi_core::runtime::block_on(recorder.applied())?;

    // Group by app
    let mut by_app: BTreeMap<String, Vec<(String, bool)>> = BTreeMap::new();
    for migration_name in graph.topological_order() {
        let is_applied = applied.contains(&("default".to_string(), migration_name.to_string()));
        let app = migration_app(migration_name);
        by_app.entry(app).or_default().push((migration_name.to_string(), is_applied));
    }

    for (app, migrations) in &by_app {
        if !args.apps.is_empty() && !args.apps.contains(app) { continue; }
        println!("{}", app);
        for (name, is_applied) in migrations {
            let mark = if *is_applied { "X" } else { " " };
            println!(" [{}] {}", mark, name);
        }
    }

    Ok(0)
}
```

## Acceptance Criteria
- [ ] `ormpi makemigrations` detects model changes and writes TOML file
- [ ] `ormpi makemigrations --dry-run` shows what would be created without writing
- [ ] `ormpi makemigrations --check` exits 1 when changes detected, 0 when none
- [ ] `ormpi migrate` applies all pending migrations with [OK] output
- [ ] `ormpi migrate --plan` shows pending migrations without executing
- [ ] `ormpi migrate --fake` marks applied without SQL
- [ ] `ormpi migrate app_name zero` reverses all app migrations
- [ ] `ormpi showmigrations` shows [X] for applied and [ ] for pending
- [ ] `ormpi version` prints version string
- [ ] Settings loaded from `ORMPI_SETTINGS_MODULE` env var or `--settings`
