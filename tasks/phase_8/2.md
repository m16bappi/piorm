## sqlmigrate Command

```rust
// ormpi-cli/src/commands/sqlmigrate.rs
#[derive(clap::Args)]
pub struct SqlmigrateArgs {
    pub app_label: String,
    pub migration_name: String,
    #[arg(long, help = "Show backwards migration SQL")]
    pub backwards: bool,
    #[arg(long)]
    pub database: Option<String>,
}

pub fn run_sqlmigrate(py: Python<'_>, args: SqlmigrateArgs) -> PyResult<i32> {
    let migration_dir = find_migrations_dir(py)?;
    let db_alias = args.database.as_deref().unwrap_or("default");

    let pool = get_pool(db_alias)?;
    let backend = QueryBackend(pool.backend());
    let graph = MigrationGraph::load_from_dir(&migration_dir)?;

    let migration_key = format!("{}.{}", args.app_label, args.migration_name);
    let migration = graph.get(&migration_key)
        .ok_or_else(|| format!("Migration '{}' not found", migration_key))?;

    let ops = migration.to_operations()?;

    println!("BEGIN;");
    println!();

    for op in &ops {
        let sqls = if args.backwards {
            op.database_backwards(&backend)
        } else {
            op.database_forwards(&backend)
        };
        let direction = if args.backwards { "BACKWARD" } else { "FORWARD" };
        println!("-- {} operation: {}", direction, op.describe());
        for sql in sqls {
            println!("{};", sql);
        }
        println!();
    }

    println!("COMMIT;");

    Ok(0)
}
```

## inspectdb Command

```rust
// ormpi-cli/src/commands/inspectdb.rs
#[derive(clap::Args)]
pub struct InspectdbArgs {
    pub tables: Vec<String>,     // specific tables; empty = all
    #[arg(long)]
    pub database: Option<String>,
    #[arg(long)]
    pub include_views: bool,
}

pub fn run_inspectdb(py: Python<'_>, args: InspectdbArgs) -> PyResult<i32> {
    let db_alias = args.database.as_deref().unwrap_or("default");
    let pool = get_pool(db_alias)?;
    let backend = QueryBackend(pool.backend());

    // Introspect schema
    let tables = ormpi_core::runtime::block_on(introspect_schema(&pool, &args.tables))?;

    println!("# This file was auto-generated by 'ormpi inspectdb'.");
    println!("# You'll have to do the following manually:");
    println!("#   * Rearrange models' order");
    println!("#   * Make sure each model has one field with primary_key=True");
    println!("from ormpi import Model, CharField, IntegerField, ForeignKey, CASCADE");
    println!("from ormpi import TextField, BooleanField, DateTimeField, FloatField");
    println!();

    for table in &tables {
        let class_name = snake_to_pascal(&table.name);
        println!("class {}(Model):", class_name);
        for col in &table.columns {
            let field = column_to_field(col);
            println!("    {} = {}", col.name, field);
        }
        println!();
        println!("    class Meta:");
        println!("        db_table = '{}'", table.name);
        println!("        managed = False  # or True if you want ormpi to manage this table");
        println!();
    }

    Ok(0)
}

/// Introspect all tables from database
async fn introspect_schema(pool: &OrmpiPool, filter: &[String]) -> crate::Result<Vec<TableInfo>> {
    match pool {
        OrmpiPool::Sqlite(p) => {
            let rows = sqlx::query(
                "SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%' ORDER BY name"
            ).fetch_all(p).await?;

            let mut tables = Vec::new();
            for row in rows {
                let name: String = row.get("name");
                if !filter.is_empty() && !filter.contains(&name) { continue; }

                let cols = sqlx::query(&format!("PRAGMA table_info({})", name))
                    .fetch_all(p).await?;

                let columns = cols.iter().map(|c| ColumnInfo {
                    name: c.get::<String, _>("name"),
                    type_str: c.get::<String, _>("type"),
                    not_null: c.get::<i32, _>("notnull") != 0,
                    pk: c.get::<i32, _>("pk") != 0,
                    default: c.get::<Option<String>, _>("dflt_value"),
                }).collect();

                tables.push(TableInfo { name, columns });
            }
            Ok(tables)
        }
        // Postgres: information_schema.columns
        // MySQL: information_schema.columns
        _ => unimplemented!("Postgres/MySQL introspection")
    }
}

fn column_to_field(col: &ColumnInfo) -> String {
    let type_str = col.type_str.to_uppercase();
    let mut kwargs = Vec::new();

    if col.not_null && !col.pk { kwargs.push("null=False".to_string()); }
    if !col.not_null { kwargs.push("null=True".to_string()); }
    if col.pk { kwargs.push("primary_key=True".to_string()); }

    let field_class = match type_str.as_str() {
        t if t.starts_with("VARCHAR") => {
            let max_len = extract_size(&type_str).unwrap_or(255);
            kwargs.push(format!("max_length={}", max_len));
            "CharField"
        }
        "TEXT"    => "TextField",
        "INTEGER" | "INT" => "IntegerField",
        "BIGINT"  => "BigIntegerField",
        "REAL" | "FLOAT" | "DOUBLE" => "FloatField",
        "BOOLEAN" | "BOOL" => "BooleanField",
        t if t.starts_with("DECIMAL") | t.starts_with("NUMERIC") => {
            let (p, s) = extract_precision(&type_str).unwrap_or((10, 2));
            kwargs.push(format!("max_digits={}, decimal_places={}", p, s));
            "DecimalField"
        }
        "DATE"      => "DateField",
        "TIME"      => "TimeField",
        "DATETIME" | "TIMESTAMP" => "DateTimeField",
        "UUID"      => "UUIDField",
        "JSON" | "JSONB" => "JSONField",
        "BLOB" | "BYTEA" => "BinaryField",
        _           => "TextField",  // fallback
    };

    if kwargs.is_empty() {
        format!("{}()", field_class)
    } else {
        format!("{}({})", field_class, kwargs.join(", "))
    }
}
```

## run_cli Export to Python

```rust
// ormpi-py/src/lib.rs (updated)
#[pyfunction]
fn run_cli(args: Vec<String>) -> PyResult<i32> {
    // Build argv with program name
    let mut argv = vec!["ormpi".to_string()];
    argv.extend(args);

    // Run via clap parse
    use ormpi_cli::Cli;
    let result = std::panic::catch_unwind(|| {
        ormpi_cli::run_with_args(&argv)
    });

    match result {
        Ok(Ok(code))  => Ok(code),
        Ok(Err(e))    => {
            eprintln!("Error: {}", e);
            Ok(1)
        }
        Err(_)        => Ok(2),
    }
}
```
