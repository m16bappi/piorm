## atomic() — Python Implementation

`atomic()` is an async context manager implemented as a Python class that wraps the Rust transaction:

```python
# python/ormpi/transaction.py
from ormpi._core import begin as _begin, PyAtomicContext

class atomic:
    """
    Async context manager for database transactions with automatic savepoint nesting.

    Usage:
        async with atomic() as txn:
            await MyModel.objects.create(name="test")
            # commits on exit
            # rolls back on exception

        # Nested (uses savepoints):
        async with atomic() as outer:
            async with atomic() as inner:
                # Uses SAVEPOINT
                ...
    """

    def __init__(self, using: str = "default"):
        self._using = using
        self._txn = None
        self._savepoint_name = None
        self._is_nested = False

    async def __aenter__(self):
        # Check if there's already an active transaction on this thread/task
        current = _get_current_transaction(self._using)
        if current is not None and current.is_open:
            # Nested: create savepoint
            self._is_nested = True
            self._txn = current
            self._savepoint_name = await _create_savepoint(current)
        else:
            # Outermost: begin real transaction
            self._is_nested = False
            self._txn = await _begin(using=self._using)
            _set_current_transaction(self._using, self._txn)
        return self._txn

    async def __aexit__(self, exc_type, exc_val, exc_tb):
        if exc_type is None:
            # Success
            if self._is_nested:
                await _release_savepoint(self._txn, self._savepoint_name)
            else:
                await self._txn.commit()
                _clear_current_transaction(self._using)
        else:
            # Exception: rollback
            if self._is_nested:
                await _rollback_to_savepoint(self._txn, self._savepoint_name)
            else:
                await self._txn.rollback()
                _clear_current_transaction(self._using)
        return False  # don't suppress exception
```

## Active Transaction Registry

To track the "current transaction" per asyncio task (equivalent of thread-local in async context):

```python
# python/ormpi/_transaction_local.py
import asyncio
from typing import Optional

_CURRENT_TRANSACTIONS: dict = {}   # task_id → {alias: transaction}

def _get_current_transaction(using: str):
    task = asyncio.current_task()
    if task is None: return None
    return _CURRENT_TRANSACTIONS.get(id(task), {}).get(using)

def _set_current_transaction(using: str, txn):
    task = asyncio.current_task()
    if task is None: return
    _CURRENT_TRANSACTIONS.setdefault(id(task), {})[using] = txn

def _clear_current_transaction(using: str):
    task = asyncio.current_task()
    if task is None: return
    _CURRENT_TRANSACTIONS.get(id(task), {}).pop(using, None)
```

## Savepoint Functions (Rust → Python)

```rust
#[pyfunction]
fn create_savepoint<'py>(py: Python<'py>, txn: &mut PyTransaction) -> PyResult<&'py PyAny> {
    let sp_name = format!("sp_{}", rand_hex(8));
    let sp_name_clone = sp_name.clone();

    // Execute SAVEPOINT {name}
    let sql = format!("SAVEPOINT {}", sp_name);
    pyo3_async_runtimes::tokio::future_into_py(py, async move {
        if let Some(ref mut inner) = txn.inner {
            inner.execute_raw(&sql).await?;
            inner.depth += 1;
        }
        Ok(sp_name_clone)
    })
}

#[pyfunction]
fn release_savepoint<'py>(py: Python<'py>, txn: &mut PyTransaction, name: String) -> PyResult<&'py PyAny> {
    let sql = format!("RELEASE SAVEPOINT {}", name);
    pyo3_async_runtimes::tokio::future_into_py(py, async move {
        if let Some(ref mut inner) = txn.inner {
            inner.execute_raw(&sql).await?;
            inner.depth = inner.depth.saturating_sub(1);
        }
        Ok(())
    })
}

#[pyfunction]
fn rollback_to_savepoint<'py>(py: Python<'py>, txn: &mut PyTransaction, name: String) -> PyResult<&'py PyAny> {
    let sql = format!("ROLLBACK TO SAVEPOINT {}", name);
    pyo3_async_runtimes::tokio::future_into_py(py, async move {
        if let Some(ref mut inner) = txn.inner {
            inner.execute_raw(&sql).await?;
        }
        Ok(())
    })
}
```

## QuerySet.using_transaction()

```rust
fn using_transaction(&self, txn: &PyTransaction) -> PyResult<Self> {
    let txn_id = txn.inner.as_ref()
        .ok_or_else(|| PyRuntimeError::new_err("Transaction already consumed"))?
        as *const OrmpiTransaction as u64;

    let mut state = self.state.clone();
    state.transaction_id = Some(txn_id);
    Ok(Self { state })
}
```

When executing with a transaction, use the transaction's connection instead of pool:

```rust
// In evaluate()
if let Some(txn_id) = state.transaction_id {
    // Get transaction by pointer (unsafe, but controlled)
    let txn_ptr = txn_id as *mut OrmpiTransaction;
    let txn = unsafe { &mut *txn_ptr };
    execute_with_transaction(txn, &sql, &values).await?
} else {
    execute_with_pool(pool, &sql, &values).await?
}
```

> Note: This approach requires that the transaction outlives the query execution, which is guaranteed by the Python context manager pattern. The `unsendable` marker on PyTransaction prevents cross-thread misuse.

## Integration Example (tests)

```python
async def test_atomic_commit():
    async with atomic() as txn:
        await Article.objects.using_transaction(txn).create(title="committed")
    # After context: committed to DB
    count = await Article.objects.count()
    assert count == 1

async def test_atomic_rollback():
    try:
        async with atomic() as txn:
            await Article.objects.using_transaction(txn).create(title="will_rollback")
            raise ValueError("oops")
    except ValueError:
        pass
    count = await Article.objects.count()
    assert count == 0  # rolled back

async def test_nested_atomic():
    async with atomic() as outer:
        await Article.objects.create(title="outer")
        try:
            async with atomic() as inner:  # SAVEPOINT
                await Article.objects.create(title="inner_fail")
                raise ValueError("inner error")
        except ValueError:
            pass  # ROLLBACK TO SAVEPOINT — outer still active
    # Only "outer" committed
    titles = await Article.objects.values_list("title", flat=True)
    assert "outer" in titles
    assert "inner_fail" not in titles
```
