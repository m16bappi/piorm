## python/ormpi/query.pyi

```python
from __future__ import annotations
from typing import Any, Generic, TypeVar, overload, Awaitable
from collections.abc import AsyncIterator
from ormpi.transaction import Transaction

M = TypeVar("M", bound="Model")

class Q:
    def __init__(self, **kwargs: Any) -> None: ...
    def __and__(self, other: Q) -> Q: ...
    def __or__(self, other: Q) -> Q: ...
    def __invert__(self) -> Q: ...

class QuerySet(Generic[M]):
    def filter(self, *args: Q, **kwargs: Any) -> QuerySet[M]: ...
    def exclude(self, *args: Q, **kwargs: Any) -> QuerySet[M]: ...
    def all(self) -> QuerySet[M]: ...
    def none(self) -> QuerySet[M]: ...
    def order_by(self, *fields: str) -> QuerySet[M]: ...
    def reverse(self) -> QuerySet[M]: ...
    def distinct(self, *fields: str) -> QuerySet[M]: ...
    def select_related(self, *fields: str, depth: int | None = None) -> QuerySet[M]: ...
    def prefetch_related(self, *lookups: str) -> QuerySet[M]: ...
    def only(self, *fields: str) -> QuerySet[M]: ...
    def defer(self, *fields: str) -> QuerySet[M]: ...
    def using(self, alias: str) -> QuerySet[M]: ...
    def using_transaction(self, txn: Transaction) -> QuerySet[M]: ...
    def annotate(self, **kwargs: Any) -> QuerySet[M]: ...
    def values(self, *fields: str) -> QuerySet[dict[str, Any]]: ...
    def values_list(self, *fields: str, flat: bool = False) -> QuerySet[tuple[Any, ...]]: ...
    def __aiter__(self) -> AsyncIterator[M]: ...
    def __await__(self) -> Any: ...
    @overload
    def __getitem__(self, key: int) -> Awaitable[M]: ...
    @overload
    def __getitem__(self, key: slice) -> QuerySet[M]: ...
    def get(self, *args: Q, **kwargs: Any) -> Awaitable[M]: ...
    def create(self, **kwargs: Any) -> Awaitable[M]: ...
    def get_or_create(self, defaults: dict[str, Any] | None = None, **kwargs: Any) -> Awaitable[tuple[M, bool]]: ...
    def update_or_create(self, defaults: dict[str, Any] | None = None, **kwargs: Any) -> Awaitable[tuple[M, bool]]: ...
    def update(self, **kwargs: Any) -> Awaitable[int]: ...
    def delete(self) -> Awaitable[tuple[int, dict[str, int]]]: ...
    def count(self) -> Awaitable[int]: ...
    def exists(self) -> Awaitable[bool]: ...
    def first(self) -> Awaitable[M | None]: ...
    def last(self) -> Awaitable[M | None]: ...
    def earliest(self, *fields: str) -> Awaitable[M]: ...
    def latest(self, *fields: str) -> Awaitable[M]: ...
    def aggregate(self, **kwargs: Any) -> Awaitable[dict[str, Any]]: ...
    def bulk_create(self, objs: list[M], batch_size: int | None = None, ignore_conflicts: bool = False) -> Awaitable[list[M]]: ...
    def bulk_update(self, objs: list[M], fields: list[str], batch_size: int | None = None) -> Awaitable[int]: ...
    def raw(self, sql: str, params: list[Any] | None = None) -> RawQuerySet[M]: ...
    def iterator(self, chunk_size: int = 2000) -> AsyncIterator[M]: ...
```
