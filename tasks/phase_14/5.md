### Expected breakage points to watch for

If any of the following errors appear, they indicate a missed rename somewhere:

| Error | Cause | Fix |
|-------|-------|-----|
| `AttributeError: '_values'` on `NoneType` | `call0()` returned `None` — `__init__` crashed | Check `__init__` slot assignment order |
| `AttributeError: _ormpi_values` | A Rust path still uses the old name | Search all `.rs` for `_ormpi_values` |
| `AttributeError: _ormpi_dirty` | A Rust path still uses the old name | Search all `.rs` for `_ormpi_dirty` |
| `AttributeError: _ormpi_state` | A Rust path still uses the old name | Search all `.rs` for `_ormpi_state` |
| `TypeError: cannot set '_values' attribute of immutable type` | `frozen` on `PyModel` is blocking a slot write from Rust | Check that `setattr` targets the *instance*, not the class |

## Part B — New correctness assertions

Add a new test file `__tests__/test_instance_layout.py` with the following assertions.
These do not require a database connection — they test pure Python-level behaviour.

```python
"""
Phase 14 — instance layout correctness tests.

These tests verify the __slots__ + bitmask dirty-tracking layout without
requiring a live database connection.
"""
import sys
import pytest
from ormpi.models import Model
from ormpi.fields import CharField, IntegerField


class Article(Model):
    title = CharField(max_length=200)
    views = IntegerField(default=0)

    class Meta:
        # Use a fake db_table so no actual DB migration is needed
        db_table = "test_article_layout"


class TestSlots:
    def test_no_dict_on_instance(self):
        """Model instances must not have __dict__ after __slots__ migration."""
        a = Article()
        assert not hasattr(a, "__dict__"), (
            "__dict__ present — __slots__ is not active. "
            "Check that PyModel has `frozen` and that models.py declares __slots__."
        )

    def test_slots_declared(self):
        assert "__slots__" in Model.__dict__
        assert set(Model.__slots__) == {"_values", "_dirty", "_adding", "_db", "__weakref__"}

    def test_field_bits_set_on_class(self):
        bits = Article._field_bits
        assert "id"    in bits
        assert "title" in bits
        assert "views" in bits
        # All bits are distinct powers of two
        values = list(bits.values())
        assert len(values) == len(set(values))
        for v in values:
            assert v > 0 and (v & (v - 1)) == 0, f"{v} is not a power of two"

    def test_weakref_support(self):
        import weakref
        a = Article()
        ref = weakref.ref(a)
        assert ref() is a


class TestDirtyTracking:
    def test_clean_after_construction_no_kwargs(self):
        a = Article()
        assert a._dirty == 0

    def test_dirty_after_kwarg_init(self):
        a = Article(title="Hello")
        assert a._dirty & Article._field_bits["title"]

    def test_dirty_after_setattr(self):
        a = Article()
        a.title = "New"
        assert a._dirty & Article._field_bits["title"]
        assert not (a._dirty & Article._field_bits["views"])

    def test_multiple_dirty_fields(self):
        a = Article()
        a.title = "A"
        a.views = 5
        expected = Article._field_bits["title"] | Article._field_bits["views"]
        assert a._dirty == expected

    def test_dirty_cleared_manually(self):
        a = Article(title="x")
        a._dirty = 0
        assert a._dirty == 0

    def test_get_dirty_field_names(self):
        a = Article(title="x", views=3)
        dirty_names = {n for n, b in Article._field_bits.items() if a._dirty & b}
        assert "title" in dirty_names
        assert "views" in dirty_names

    def test_slot_writes_do_not_set_dirty(self):
        """Writing to _values/_dirty/_adding/_db must not recurse or set dirty bits."""
        a = Article()
        a._dirty = 0b1111
        a._adding = False
        a._db = "replica"
        # These are slot writes — should not affect _dirty
        assert a._dirty == 0b1111
        assert a._adding == False
        assert a._db == "replica"


class TestGetattr:
    def test_field_value_readable_via_attribute(self):
        a = Article()
        a._values["title"] = "direct"
        assert a.title == "direct"

    def test_missing_field_raises_attribute_error(self):
        a = Article()
        with pytest.raises(AttributeError):
            _ = a.nonexistent_field

    def test_pk_property(self):
        a = Article()
        a._values["id"] = 42
        assert a.pk == 42


class TestRepr:
    def test_repr_shows_pk(self):
        a = Article()
        a._values["id"] = 7
        assert repr(a) == "<Article: pk=7>"

    def test_repr_unknown_pk(self):
        a = Article()
        assert repr(a) == "<Article: pk=?>"
```

## Part C — Memory benchmark

Add `benchmark/bench_instance_layout.py`. This file is standalone (no DB needed) and
measures the per-instance overhead before and after Phase 14. Run it manually; it is not
part of the pytest suite.

```python
"""
benchmark/bench_instance_layout.py

Measures per-instance memory overhead of ormpi model instances.
Run with: python benchmark/bench_instance_layout.py

Expected output (Phase 14 target):
    100,000 instances: X.X MB overhead  (target: ≤ 10 MB)
"""
import sys
import gc
import tracemalloc

# Must import after ormpi is on the path
sys.path.insert(0, "python")
import ormpi  # noqa: E402 — path setup required first
from ormpi.models import Model
from ormpi.fields import CharField, IntegerField


class Article(Model):
    title  = CharField(max_length=200)
    views  = IntegerField(default=0)

    class Meta:
        db_table = "bench_article"


N = 100_000

gc.collect()
gc.disable()
tracemalloc.start()

snapshot_before = tracemalloc.take_snapshot()
instances = [Article(title=f"title_{i}", views=i) for i in range(N)]
snapshot_after = tracemalloc.take_snapshot()

tracemalloc.stop()
gc.enable()

stats = snapshot_after.compare_to(snapshot_before, "lineno")
total_bytes = sum(s.size_diff for s in stats)

overhead_per = total_bytes / N
total_mb = total_bytes / (1024 * 1024)

print(f"\n{'─'*50}")
print(f"  Instances created : {N:,}")
print(f"  Total heap delta  : {total_mb:.2f} MB")
print(f"  Overhead per inst : {overhead_per:.0f} bytes")
print(f"  Target overhead   : ≤ 72 bytes/instance")
print(f"  Status            : {'✓ PASS' if overhead_per <= 150 else '✗ FAIL'}")
print(f"{'─'*50}\n")

# Also verify __dict__ is absent
sample = instances[0]
has_dict = hasattr(sample, "__dict__")
print(f"  __dict__ present  : {has_dict}  (should be False)")
print(f"  __slots__ active  : {not has_dict}")
print()
```

### Target numbers

| Metric | Before Phase 14 | After Phase 14 | Pass threshold |
|--------|-----------------|----------------|----------------|
| Overhead / instance | ~330 bytes | ~72 bytes | ≤ 150 bytes |
| Total for 100K | ~33 MB | ~7.2 MB | ≤ 10 MB |
| `__dict__` present | Yes | No | No |

## Part D — Dirty-tracking microbenchmark

Add to `benchmark/bench_instance_layout.py` after the memory section:

```python
import timeit

# ── Dirty bit set ──────────────────────────────────────────────────────────
N_OPS = 1_000_000

a = Article()
title_bit = Article._field_bits["title"]

# New: bitmask OR
t_bitmask = timeit.timeit(
    "a._dirty |= title_bit",
    globals={"a": a, "title_bit": title_bit},
    number=N_OPS,
)

print(f"  Bitmask OR  ({N_OPS:,} ops): {t_bitmask:.3f}s  "
      f"({t_bitmask/N_OPS*1e9:.1f} ns/op)")

# Baseline comparison: set.add (simulate old behaviour)
old_dirty = set()
t_set = timeit.timeit(
    "old_dirty.add('title')",
    globals={"old_dirty": old_dirty},
    number=N_OPS,
)

print(f"  Set add     ({N_OPS:,} ops): {t_set:.3f}s  "
      f"({t_set/N_OPS*1e9:.1f} ns/op)")
print(f"  Speedup     : {t_set/t_bitmask:.1f}×")
```

Expected speedup: 3–5× for the bitmask OR over set.add.
