# Task 12.1 — Restructure Python Package Layout

## Objective

Reorganise `python/ormpi/` into a clean modular namespace with no backward-compatibility
shims. Old flat import paths are removed. Files that existed only for compatibility
(`aggregates.py`, `aggregates.pyi`, `db/`) are deleted. The only supported import paths
after this task are the canonical ones listed in the acceptance criteria below.

---

### `python/ormpi/exceptions.py` (canonical — already exists, update in place)

```python
"""
Exception hierarchy for ormpi.

All exceptions are also accessible via `ormpi.exceptions.*`.
"""

__all__ = [
    "OrmpiError",
    "DatabaseError",
    "IntegrityError",
    "OperationalError",
    "ValidationError",
    "FieldError",
    "TransactionError",
    "MigrationError",
    "DoesNotExist",
    "MultipleObjectsReturned",
    "ImproperlyConfigured",
]


class OrmpiError(Exception):
    """Base class for all ormpi exceptions."""


class DatabaseError(OrmpiError):
    """A database-level error occurred."""


class IntegrityError(DatabaseError):
    """A constraint (UNIQUE, FK, NOT NULL) was violated."""


class OperationalError(DatabaseError):
    """The database reported an operational error (connection lost, syntax, etc.)."""


class ValidationError(OrmpiError):
    """A field value failed validation."""
    message: str

    def __init__(self, message: str) -> None:
        self.message = message
        super().__init__(message)


class FieldError(OrmpiError):
    """An invalid field name or lookup was used."""


class TransactionError(OrmpiError):
    """A transaction operation failed (e.g., commit on a closed transaction)."""


class MigrationError(OrmpiError):
    """Migration detection, generation, or application failed."""


class DoesNotExist(OrmpiError):
    """Raised when `.get()` matches no rows."""


class MultipleObjectsReturned(OrmpiError):
    """Raised when `.get()` matches more than one row."""


class ImproperlyConfigured(OrmpiError):
    """ormpi was not configured correctly (e.g., configure() not called)."""
```

The file already exists and is already canonical — just update docstrings and `__all__` as shown above.

### `python/ormpi/expr.py` (new)

```python
"""
Query expression objects: Q combinators, F references, and aggregate functions.

    from ormpi.expr import Q, F, Count, Sum, Avg, Max, Min
    from ormpi import expr   # namespace access: expr.Q(...)
"""

__all__ = [
    "Q",
    "F",
    "Aggregate",
    "Count",
    "Sum",
    "Avg",
    "Max",
    "Min",
]

try:
    from ormpi._core import Q, Aggregate, Count, Sum, Avg, Max, Min  # type: ignore[attr-defined]
except ImportError:
    # Pure-Python fallback used during stub-only / CI type-check runs
    class Q:  # type: ignore[no-redef]
        def __init__(self, **kwargs):  ...
        def __and__(self, other): return other
        def __or__(self, other): return other
        def __invert__(self): return self

    class Aggregate:  # type: ignore[no-redef]
        def __init__(self, field: str, **kwargs): ...

    class Count(Aggregate): ...   # type: ignore[no-redef]
    class Sum(Aggregate): ...     # type: ignore[no-redef]
    class Avg(Aggregate): ...     # type: ignore[no-redef]
    class Max(Aggregate): ...     # type: ignore[no-redef]
    class Min(Aggregate): ...     # type: ignore[no-redef]


class F:
    """Reference to a model field for use in annotations and updates.

    Example::

        # Increment a counter atomically
        await Post.objects.filter(pk=1).update(views=F("views") + 1)
    """

    def __init__(self, name: str) -> None:
        self.name = name

    def __repr__(self) -> str:
        return f"F({self.name!r})"

    # Arithmetic operators — return an F-expression node
    def __add__(self, other):      return _FExpr(self, "+", other)
    def __radd__(self, other):     return _FExpr(other, "+", self)
    def __sub__(self, other):      return _FExpr(self, "-", other)
    def __rsub__(self, other):     return _FExpr(other, "-", self)
    def __mul__(self, other):      return _FExpr(self, "*", other)
    def __rmul__(self, other):     return _FExpr(other, "*", self)
    def __truediv__(self, other):  return _FExpr(self, "/", other)
    def __mod__(self, other):      return _FExpr(self, "%", other)
    def __neg__(self):             return _FExpr(None, "-", self)


class _FExpr:
    """Internal: represents a binary expression built from F() objects."""
    __slots__ = ("lhs", "op", "rhs")

    def __init__(self, lhs, op: str, rhs) -> None:
        self.lhs = lhs
        self.op  = op
        self.rhs = rhs

    def __repr__(self) -> str:
        return f"({self.lhs!r} {self.op} {self.rhs!r})"

    # Allow chaining: F("x") + 1 + 2
    def __add__(self, other):  return _FExpr(self, "+", other)
    def __sub__(self, other):  return _FExpr(self, "-", other)
    def __mul__(self, other):  return _FExpr(self, "*", other)
```

### `python/ormpi/db.py` (update — flatten `db/` sub-package into single file)

```python
"""
Database connection utilities.

    from ormpi.db import connections, configure, close
    from ormpi import db   # namespace access: db.connections[...]
"""

__all__ = [
    "connections",
    "configure",
    "close",
    "create_table",
    "create_all_tables",
]

from ormpi._core import ConnectionProxy as _ConnectionProxy
from ormpi._core import create_all_tables as _create_all_tables
from ormpi._core import create_table as _create_table
from ormpi._core import configure as _configure_core
from ormpi._core import close as _close_core

#: Proxy for dict-style access to connection wrappers by alias.
#: Example: ``ormpi.db.connections["default"].vendor``
connections: _ConnectionProxy = _ConnectionProxy()


async def configure(
    databases: dict,
    *,
    auto_sync: bool = False,
) -> None:
    """Configure database connections.

    Thin wrapper around the Rust ``_core.configure`` that is also accessible
    as ``ormpi.configure()`` from the root package.

    Args:
        databases: Mapping of alias → connection config dict.
        auto_sync: When ``True``, create all model tables automatically
                   using ``CREATE TABLE IF NOT EXISTS`` after pools are ready.
    """
    await _configure_core(databases, auto_sync=auto_sync)


async def close() -> None:
    """Close all database connection pools."""
    await _close_core()


async def create_table(model_name: str, alias: str = "default") -> None:
    """Create the database table for a single registered model."""
    await _create_table(model_name, alias)


async def create_all_tables(alias: str = "default") -> None:
    """Create tables for all registered models on the given database alias."""
    await _create_all_tables(alias)
```

Delete `python/ormpi/db/__init__.py` and the `db/` directory after moving to `db.py`.

## Canonical Import Paths (the ONLY supported paths after this task)

| Symbol(s) | Import from |
|-----------|-------------|
| `Model` | `ormpi` or `ormpi.models` |
| `configure`, `close` | `ormpi` or `ormpi.db` |
| `CharField`, `IntegerField`, `ForeignKey`, `CASCADE`, … | `ormpi.fields` |
| `Q`, `F`, `Count`, `Sum`, `Avg`, `Max`, `Min` | `ormpi.expr` |
| `atomic`, `begin`, `Transaction` | `ormpi.transaction` |
| `connections`, `create_table`, `create_all_tables` | `ormpi.db` |
| `OrmpiError`, `DoesNotExist`, … | `ormpi.exceptions` |
| `DatabaseRouter` | `ormpi.routers` |

Root convenience (also valid — but prefer sub-module for new code):
`from ormpi import Q, F, Model, configure, close, CharField, IntegerField, …`

**Everything else is removed.** No `ormpi.aggregates`, no `db/` sub-package.
