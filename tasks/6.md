# Task 6 — `@atomic()` Decorator + `propagation="requires_new"`

## Background

The doc prescribes two Python transaction styles:

> 2. **`@atomic()` decorator + `ContextVar` ambient** — zero-signature-change style
>    ```python
>    @ormpi.atomic()
>    async def transfer(from_id, to_id, amount):
>        src = await Account.objects.select_for_update().get(id=from_id)
>        ...
>        await src.asave()   # picks up ambient txn
>        await dst.asave()   # same txn
>    ```
>
> 4. **`REQUIRES_NEW` escape hatch** — for audit logs and notifications that
>    must commit independently:
>    ```python
>    @ormpi.atomic(propagation="requires_new")
>    async def log_event(event):
>        await AuditLog.create(**event)
>    ```

Currently `atomic` is only a context manager class. This task adds:

1. Decorator support — `@atomic()` wrapping an async function.
2. `propagation` parameter with two modes:
   - `"default"` (existing behaviour) — join existing transaction via SAVEPOINT
     if one is active; begin a new transaction otherwise.
   - `"requires_new"` — always begin a **new, independent** transaction,
     suspending (but not rolling back) any currently active one.

## Design

```python
import functools, inspect

class atomic:
    def __init__(
        self,
        using: str = "default",
        propagation: str = "default",  # "default" | "requires_new"
    ) -> None:
        self._using = using
        self._propagation = propagation
        ...

    def __call__(self, func):
        """Enable use as @atomic() decorator on async functions."""
        if not inspect.iscoroutinefunction(func):
            raise TypeError("@atomic() can only decorate async functions")

        @functools.wraps(func)
        async def wrapper(*args, **kwargs):
            async with atomic(using=self._using, propagation=self._propagation):
                return await func(*args, **kwargs)

        return wrapper

    async def __aenter__(self) -> Transaction:
        if self._propagation == "requires_new":
            # Suspend any current transaction (don't touch it, just hide it)
            self._suspended_token = _suspend_current_transaction(self._using)
            self._is_nested = False
            self._txn = await _begin(using=self._using)
            self._ctx_token = _set_current_transaction(self._using, self._txn)
        else:
            # Existing default behaviour (from Task 5)
            ...
        return self._txn
```

For `requires_new`, `_suspend_current_transaction` saves and clears the ambient
`ContextVar` so the new inner transaction starts clean. On `__aexit__`, the
suspended transaction is restored.

## Test Cases

### Tests — `__tests__/transactions/test_atomic_decorator.py`

**TC-6.1 — `@atomic()` decorator commits on success**
```python
from ormpi.transaction import atomic

@atomic()
async def create_author_decorated(name, email):
    await Author.objects.create(name=name, email=email)

@pytest.mark.asyncio
async def test_decorator_commits_on_success():
    await create_author_decorated("Deco", "deco@test.com")
    assert await Author.objects.filter(email="deco@test.com").count() == 1
```

**TC-6.2 — `@atomic()` decorator rolls back on exception**
```python
@atomic()
async def create_and_fail(name, email):
    await Author.objects.create(name=name, email=email)
    raise ValueError("nope")

@pytest.mark.asyncio
async def test_decorator_rolls_back_on_exception():
    with pytest.raises(ValueError):
        await create_and_fail("WillFail", "wf@test.com")
    assert await Author.objects.filter(email="wf@test.com").count() == 0
```

**TC-6.3 — `@atomic()` return value is propagated**
```python
@atomic()
async def get_count():
    return await Author.objects.count()

@pytest.mark.asyncio
async def test_decorator_propagates_return():
    await Author.objects.create(name="One", email="one@test.com")
    count = await get_count()
    assert count == 1
```

**TC-6.4 — `@atomic()` on sync function raises `TypeError`**
```python
def test_decorator_rejects_sync_function():
    with pytest.raises(TypeError, match="async"):
        @atomic()
        def sync_fn():
            pass
```

**TC-6.5 — Nested `@atomic()` decorated functions use SAVEPOINT**
```python
@atomic()
async def inner_op():
    await Author.objects.create(name="Inner", email="inner@test.com")
    raise ValueError("inner fail")

@atomic()
async def outer_op():
    await Author.objects.create(name="Outer", email="outer@test.com")
    try:
        await inner_op()  # savepoint — rolled back
    except ValueError:
        pass  # outer continues

@pytest.mark.asyncio
async def test_nested_decorated_savepoint():
    await outer_op()
    assert await Author.objects.filter(email="outer@test.com").count() == 1
    assert await Author.objects.filter(email="inner@test.com").count() == 0
```

### Tests — `__tests__/transactions/test_requires_new.py`

**TC-6.6 — `requires_new` always opens a fresh transaction**
```python
@pytest.mark.asyncio
async def test_requires_new_independent_of_outer():
    @atomic(propagation="requires_new")
    async def log_event(name):
        await Author.objects.create(name=name, email=f"{name}@log.com")

    with pytest.raises(ValueError):
        async with ormpi.transaction.atomic() as outer:
            await Author.objects.create(name="Main", email="main@test.com")
            await log_event("AuditLog")  # commits independently
            raise ValueError("outer fails")

    # Outer rolled back
    assert await Author.objects.filter(email="main@test.com").count() == 0
    # Inner (requires_new) committed independently
    assert await Author.objects.filter(email="AuditLog@log.com").count() == 1
```

**TC-6.7 — `requires_new` does NOT inherit outer ambient transaction**
```python
@pytest.mark.asyncio
async def test_requires_new_no_ambient_inheritance():
    from ormpi._transaction_local import _get_current_transaction

    @atomic(propagation="requires_new")
    async def check_ambient():
        # Should have its own fresh txn, not the outer one
        inner_txn = _get_current_transaction("default")
        return inner_txn

    async with ormpi.transaction.atomic() as outer:
        inner = await check_ambient()
        assert inner is not outer
        assert inner is not None
        assert inner.is_open
```

**TC-6.8 — `requires_new` rolls back its own transaction on exception**
```python
@pytest.mark.asyncio
async def test_requires_new_own_rollback():
    @atomic(propagation="requires_new")
    async def fail_log():
        await Author.objects.create(name="FailLog", email="fl@test.com")
        raise RuntimeError("log failure")

    with pytest.raises(RuntimeError):
        await fail_log()

    assert await Author.objects.filter(email="fl@test.com").count() == 0
```

**TC-6.9 — `requires_new` outer ambient is restored after inner exits**
```python
@pytest.mark.asyncio
async def test_requires_new_outer_ambient_restored():
    from ormpi._transaction_local import _get_current_transaction

    @atomic(propagation="requires_new")
    async def side_effect():
        pass

    async with ormpi.transaction.atomic() as outer:
        await side_effect()
        restored = _get_current_transaction("default")
        assert restored is outer
```
