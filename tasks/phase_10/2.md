## .github/workflows/ci.yml

```yaml
name: CI

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]

env:
  CARGO_TERM_COLOR: always
  RUST_BACKTRACE: 1

jobs:
  # ─── Rust unit tests ─────────────────────────────────────────────────────────
  rust-tests:
    name: Rust Tests
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions-rust-lang/setup-rust-toolchain@v1
        with: { toolchain: stable, components: "clippy,rustfmt" }
      - run: cargo test --workspace --all-features
      - run: cargo clippy --workspace --all-features -- -D warnings
      - run: cargo fmt --check

  # ─── Python integration tests (SQLite) ───────────────────────────────────────
  test-sqlite:
    name: Python Tests (SQLite) / Python ${{ matrix.python-version }}
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        python-version: ["3.9", "3.10", "3.11", "3.12", "3.13"]
    steps:
      - uses: actions/checkout@v4
        with: { submodules: true }
      - uses: actions/setup-python@v5
        with: { python-version: "${{ matrix.python-version }}" }
      - uses: PyO3/maturin-action@v1
        with:
          command: develop
          args: --extras test
      - run: pytest tests/integration/ tests/unit/ -v --tb=short

  # ─── Postgres integration tests ──────────────────────────────────────────────
  test-postgres:
    name: Python Tests (Postgres)
    runs-on: ubuntu-latest
    services:
      postgres:
        image: postgres:16-alpine
        env:
          POSTGRES_USER: ormpi
          POSTGRES_PASSWORD: testpass
          POSTGRES_DB: ormpi_test
        ports: ["5432:5432"]
        options: >-
          --health-cmd pg_isready
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5
    env:
      TEST_POSTGRES: "1"
      POSTGRES_HOST: localhost
      POSTGRES_USER: ormpi
      POSTGRES_PASSWORD: testpass
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with: { python-version: "3.12" }
      - uses: PyO3/maturin-action@v1
        with: { command: develop, args: "--extras test" }
      - run: pytest tests/integration/ -v --tb=short

  # ─── MySQL integration tests ─────────────────────────────────────────────────
  test-mysql:
    name: Python Tests (MySQL)
    runs-on: ubuntu-latest
    services:
      mysql:
        image: mysql:8-oracle
        env:
          MYSQL_ROOT_PASSWORD: testpass
          MYSQL_DATABASE: ormpi_test
        ports: ["3306:3306"]
        options: >-
          --health-cmd "mysqladmin ping"
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5
    env:
      TEST_MYSQL: "1"
      MYSQL_HOST: localhost
      MYSQL_USER: root
      MYSQL_PASSWORD: testpass
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with: { python-version: "3.12" }
      - uses: PyO3/maturin-action@v1
        with: { command: develop, args: "--extras test" }
      - run: pytest tests/integration/ -v --tb=short

  # ─── Type checking ───────────────────────────────────────────────────────────
  type-check:
    name: Type Check (mypy + pyright)
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with: { python-version: "3.12" }
      - uses: PyO3/maturin-action@v1
        with: { command: develop, args: "--extras dev" }
      - run: mypy --strict python/ormpi/
      - run: pip install pyright && pyright python/ormpi/
```

## .github/workflows/release.yml

```yaml
name: Release

on:
  push:
    tags: ["v*"]

jobs:
  build-wheels:
    name: Build Wheels / ${{ matrix.os }}
    runs-on: ${{ matrix.os }}
    strategy:
      matrix:
        os: [ubuntu-latest, macos-latest, windows-latest]
    steps:
      - uses: actions/checkout@v4
      - uses: PyO3/maturin-action@v1
        with:
          command: build
          args: --release --out dist --find-interpreter
          # On Linux, uses manylinux container automatically
          manylinux: auto
      - uses: actions/upload-artifact@v4
        with:
          name: wheels-${{ matrix.os }}
          path: dist/*.whl

  build-sdist:
    name: Build Source Distribution
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: PyO3/maturin-action@v1
        with: { command: sdist, args: "--out dist" }
      - uses: actions/upload-artifact@v4
        with: { name: sdist, path: dist/*.tar.gz }

  publish:
    name: Publish to PyPI
    needs: [build-wheels, build-sdist]
    runs-on: ubuntu-latest
    environment: pypi
    permissions:
      id-token: write   # for OIDC trusted publishing
    steps:
      - uses: actions/download-artifact@v4
        with: { pattern: "wheels-*", merge-multiple: true, path: dist/ }
      - uses: actions/download-artifact@v4
        with: { name: sdist, path: dist/ }
      - uses: pypa/gh-action-pypi-publish@release/v1
```

## Performance Benchmarks

```python
# tests/benchmarks/bench_insert.py
import asyncio
import pytest
from tests.conftest import Author

@pytest.fixture
async def populated_db(db):
    authors = [Author(name=f"A{i}", email=f"a{i}@test.com") for i in range(1000)]
    await Author.objects.bulk_create(authors)
    yield

@pytest.mark.benchmark(group="insert")
async def test_single_insert_throughput(benchmark, db):
    i = 0
    async def insert_one():
        nonlocal i; i += 1
        await Author.objects.create(name=f"Test{i}", email=f"t{i}@test.com")
    await benchmark.pedantic(insert_one, iterations=100, rounds=3)

@pytest.mark.benchmark(group="select")
async def test_full_table_scan(benchmark, populated_db):
    async def scan():
        return await Author.objects.all()
    result = await benchmark(scan)
    assert len(result) == 1000

@pytest.mark.benchmark(group="filter")
async def test_filtered_query(benchmark, populated_db):
    async def query():
        return await Author.objects.filter(name__startswith="A5")
    await benchmark(query)
```
