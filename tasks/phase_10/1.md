```python
async def test_create_and_retrieve(db):
    author = await Author.objects.create(name="Alice", email="alice@test.com")
    assert author.pk is not None
    assert author.name == "Alice"

    retrieved = await Author.objects.get(pk=author.pk)
    assert retrieved.name == "Alice"
    assert retrieved.email == "alice@test.com"

async def test_save_new(db):
    author = Author(name="Bob", email="bob@test.com")
    assert author.pk is None
    await author.save()
    assert author.pk is not None
    assert author._ormpi_state["adding"] is False

async def test_save_update(db):
    author = await Author.objects.create(name="Charlie", email="charlie@test.com")
    author.name = "Charles"
    await author.save()
    refreshed = await Author.objects.get(pk=author.pk)
    assert refreshed.name == "Charles"

async def test_save_update_only_dirty(db, monkeypatch):
    """Verify UPDATE only includes changed fields"""
    author = await Author.objects.create(name="Dave", email="dave@test.com", age=30)
    sql_calls = []
    original_execute = ormpi.db.connections["default"].execute
    monkeypatch.setattr(ormpi.db.connections["default"], "execute",
        lambda sql, *a, **kw: sql_calls.append(sql) or original_execute(sql, *a, **kw))

    author.name = "David"
    await author.save()
    assert any("name" in s for s in sql_calls)
    assert not any("age" in s for s in sql_calls)  # age not dirty

async def test_delete(db):
    author = await Author.objects.create(name="Eve", email="eve@test.com")
    pk = author.pk
    count, breakdown = await author.delete()
    assert count == 1
    assert "Author" in breakdown
    with pytest.raises(Author.DoesNotExist):
        await Author.objects.get(pk=pk)

async def test_does_not_exist(db):
    with pytest.raises(Author.DoesNotExist):
        await Author.objects.get(pk=999999)

async def test_multiple_objects_returned(db):
    await Author.objects.create(name="A", email="a@test.com")
    await Author.objects.create(name="B", email="b@test.com")
    with pytest.raises(ormpi.MultipleObjectsReturned):
        await Author.objects.get(active=True)
```

```python
async def test_filter_exact(db):
    await Author.objects.create(name="Alice", email="a@test.com")
    await Author.objects.create(name="Bob", email="b@test.com")
    results = await Author.objects.filter(name="Alice")
    assert len(results) == 1
    assert results[0].name == "Alice"

async def test_filter_icontains(db):
    await Author.objects.create(name="Alice Smith", email="a@test.com")
    await Author.objects.create(name="Bob Jones", email="b@test.com")
    results = await Author.objects.filter(name__icontains="smith")
    assert len(results) == 1

async def test_filter_in(db):
    a1 = await Author.objects.create(name="A", email="a@test.com")
    a2 = await Author.objects.create(name="B", email="b@test.com")
    a3 = await Author.objects.create(name="C", email="c@test.com")
    results = await Author.objects.filter(pk__in=[a1.pk, a3.pk])
    assert len(results) == 2

async def test_filter_gt_lt(db):
    await Author.objects.create(name="Young", email="y@test.com", age=20)
    await Author.objects.create(name="Middle", email="m@test.com", age=35)
    await Author.objects.create(name="Old", email="o@test.com", age=50)
    results = await Author.objects.filter(age__gte=30, age__lt=50)
    assert len(results) == 1
    assert results[0].name == "Middle"

async def test_q_or(db):
    await Author.objects.create(name="Alice", email="a@test.com")
    await Author.objects.create(name="Bob", email="b@test.com")
    from ormpi import Q
    results = await Author.objects.filter(Q(name="Alice") | Q(name="Bob"))
    assert len(results) == 2

async def test_q_not(db):
    await Author.objects.create(name="Alice", email="a@test.com", active=True)
    await Author.objects.create(name="Bob", email="b@test.com", active=False)
    from ormpi import Q
    results = await Author.objects.filter(~Q(active=True))
    assert len(results) == 1
    assert results[0].name == "Bob"

async def test_order_by(db):
    await Author.objects.create(name="Charlie", email="c@test.com", age=30)
    await Author.objects.create(name="Alice", email="a@test.com", age=25)
    await Author.objects.create(name="Bob", email="b@test.com", age=28)
    results = await Author.objects.order_by("age")
    assert [r.name for r in results] == ["Alice", "Bob", "Charlie"]

async def test_values(db):
    await Author.objects.create(name="Alice", email="a@test.com")
    results = await Author.objects.values("name", "email")
    assert all(isinstance(r, dict) for r in results)
    assert "name" in results[0]

async def test_count(db):
    await Author.objects.create(name="A", email="a@test.com")
    await Author.objects.create(name="B", email="b@test.com")
    count = await Author.objects.count()
    assert count == 2

async def test_exists(db):
    assert not await Author.objects.exists()
    await Author.objects.create(name="A", email="a@test.com")
    assert await Author.objects.exists()

async def test_bulk_create(db):
    authors = [Author(name=f"Author {i}", email=f"a{i}@test.com") for i in range(100)]
    result = await Author.objects.bulk_create(authors)
    assert await Author.objects.count() == 100

async def test_slicing(db):
    for i in range(20):
        await Author.objects.create(name=f"Author {i:02d}", email=f"a{i}@test.com")
    page = await Author.objects.order_by("name")[:5]
    assert len(page) == 5

async def test_update(db):
    await Author.objects.create(name="Alice", email="a@test.com", active=True)
    await Author.objects.create(name="Bob", email="b@test.com", active=True)
    count = await Author.objects.filter(active=True).update(active=False)
    assert count == 2
    assert await Author.objects.filter(active=True).count() == 0
```

## tests/integration/test_transactions.py

```python
from ormpi import atomic, begin

async def test_atomic_commit(db):
    async with atomic():
        await Author.objects.create(name="A", email="a@test.com")
    assert await Author.objects.count() == 1

async def test_atomic_rollback(db):
    with pytest.raises(ValueError):
        async with atomic():
            await Author.objects.create(name="B", email="b@test.com")
            raise ValueError("oops")
    assert await Author.objects.count() == 0

async def test_nested_atomic_partial_rollback(db):
    async with atomic():
        await Author.objects.create(name="outer", email="o@test.com")
        with pytest.raises(ValueError):
            async with atomic():
                await Author.objects.create(name="inner", email="i@test.com")
                raise ValueError("inner fail")
    # Only outer committed
    names = await Author.objects.values_list("name", flat=True)
    assert "outer" in names
    assert "inner" not in names

async def test_on_commit(db):
    fired = []
    async with atomic() as txn:
        await Author.objects.create(name="A", email="a@test.com")
        txn.on_commit(lambda: fired.append(True))
    assert len(fired) == 1

async def test_on_commit_not_fired_on_rollback(db):
    fired = []
    with pytest.raises(ValueError):
        async with atomic() as txn:
            txn.on_commit(lambda: fired.append(True))
            raise ValueError
    assert len(fired) == 0
```
