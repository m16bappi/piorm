# Task 1.6 — Row Deserialization & Value Type System

## Objective
Implement bidirectional conversion between sqlx `Row` types and Rust `Value` / Python objects. This is the foundation for all query result handling — every row fetched from the DB must be converted to Python-usable data.

## Location
- `ormpi-core/src/value.rs`
- `ormpi-core/src/row.rs`

## OrmpiValue Enum

We define our own `Value` enum (independent of sea-query's `Value`) for clarity and control:

```rust
// ormpi-core/src/value.rs
use chrono::{DateTime, NaiveDate, NaiveDateTime, NaiveTime, Utc};
use uuid::Uuid;
use serde_json::Value as JsonValue;

/// A database column value, backend-agnostic.
#[derive(Debug, Clone, PartialEq)]
pub enum OrmpiValue {
    Null,
    Bool(bool),
    Int(i64),
    Float(f64),
    Text(String),
    Bytes(Vec<u8>),
    Date(NaiveDate),
    Time(NaiveTime),
    DateTime(NaiveDateTime),
    DateTimeTz(DateTime<Utc>),
    Uuid(Uuid),
    Json(JsonValue),
    Decimal(rust_decimal::Decimal),
}

impl OrmpiValue {
    /// Convert to sea_query::Value for use in query parameters.
    pub fn to_sea_value(&self) -> sea_query::Value {
        match self {
            OrmpiValue::Null        => sea_query::Value::from(None::<i32>),
            OrmpiValue::Bool(b)     => sea_query::Value::from(*b),
            OrmpiValue::Int(i)      => sea_query::Value::from(*i),
            OrmpiValue::Float(f)    => sea_query::Value::from(*f),
            OrmpiValue::Text(s)     => sea_query::Value::from(s.as_str()),
            OrmpiValue::Bytes(b)    => sea_query::Value::from(b.as_slice()),
            OrmpiValue::Uuid(u)     => sea_query::Value::from(*u),
            OrmpiValue::Json(j)     => sea_query::Value::from(j.clone()),
            // ... etc
        }
    }
}
```

## Row → OrmpiRow (generic)

```rust
// ormpi-core/src/row.rs

/// A single database row as an ordered map of column name → value.
#[derive(Debug, Clone)]
pub struct OrmpiRow {
    pub columns: Vec<String>,
    pub values: IndexMap<String, OrmpiValue>,
}

impl OrmpiRow {
    pub fn get(&self, column: &str) -> Option<&OrmpiValue> {
        self.values.get(column)
    }

    pub fn get_required(&self, column: &str) -> crate::Result<&OrmpiValue> {
        self.values.get(column).ok_or_else(|| OrmpiError::FieldError {
            model: "?".to_string(),
            field: column.to_string(),
            message: format!("Column '{}' not found in result row", column),
        })
    }
}
```

## SQLite Row Conversion

```rust
pub fn sqlite_row_to_ormpi(row: &sqlx::sqlite::SqliteRow) -> crate::Result<OrmpiRow> {
    use sqlx::Column;
    use sqlx::Row;
    use sqlx::TypeInfo;

    let mut values = IndexMap::new();
    let mut columns = Vec::new();

    for col in row.columns() {
        let name = col.name().to_string();
        let type_name = col.type_info().name();

        let value = match type_name {
            "INTEGER" | "INT"  => row.try_get::<Option<i64>, _>(col.ordinal())
                .map(|v| v.map(OrmpiValue::Int).unwrap_or(OrmpiValue::Null))?,
            "REAL"             => row.try_get::<Option<f64>, _>(col.ordinal())
                .map(|v| v.map(OrmpiValue::Float).unwrap_or(OrmpiValue::Null))?,
            "TEXT"             => row.try_get::<Option<String>, _>(col.ordinal())
                .map(|v| v.map(OrmpiValue::Text).unwrap_or(OrmpiValue::Null))?,
            "BLOB"             => row.try_get::<Option<Vec<u8>>, _>(col.ordinal())
                .map(|v| v.map(OrmpiValue::Bytes).unwrap_or(OrmpiValue::Null))?,
            "BOOLEAN"          => row.try_get::<Option<bool>, _>(col.ordinal())
                .map(|v| v.map(OrmpiValue::Bool).unwrap_or(OrmpiValue::Null))?,
            "NULL"             => OrmpiValue::Null,
            _                  => {
                // Fallback: try as text
                row.try_get::<Option<String>, _>(col.ordinal())
                    .map(|v| v.map(OrmpiValue::Text).unwrap_or(OrmpiValue::Null))
                    .unwrap_or(OrmpiValue::Null)
            }
        };

        columns.push(name.clone());
        values.insert(name, value);
    }

    Ok(OrmpiRow { columns, values })
}
```

## OrmpiValue → PyObject

```rust
// ormpi-py/src/convert.rs
use pyo3::prelude::*;
use pyo3::types::*;

pub fn value_to_py(py: Python<'_>, value: &OrmpiValue) -> PyObject {
    match value {
        OrmpiValue::Null          => py.None(),
        OrmpiValue::Bool(b)       => b.into_py(py),
        OrmpiValue::Int(i)        => i.into_py(py),
        OrmpiValue::Float(f)      => f.into_py(py),
        OrmpiValue::Text(s)       => s.into_py(py),
        OrmpiValue::Bytes(b)      => PyBytes::new(py, b).into(),
        OrmpiValue::Date(d)       => {
            // Return Python datetime.date object
            let datetime = py.import("datetime")?;
            datetime.getattr("date")?.call1((d.year(), d.month(), d.day()))?.into()
        }
        OrmpiValue::DateTime(dt)  => {
            let datetime = py.import("datetime")?;
            datetime.getattr("datetime")?.call1((
                dt.year(), dt.month(), dt.day(),
                dt.hour(), dt.minute(), dt.second()
            ))?.into()
        }
        OrmpiValue::Uuid(u)       => u.to_string().into_py(py),
        OrmpiValue::Json(j)       => {
            // Convert via json.loads(json_string)
            let json_mod = py.import("json")?;
            json_mod.getattr("loads")?.call1((j.to_string(),))?.into()
        }
        // ... other variants
    }
}
```
