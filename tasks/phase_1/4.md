## DatabaseConfig

```rust
// ormpi-core/src/config.rs
use std::time::Duration;

#[derive(Debug, Clone)]
pub struct DatabaseConfig {
    pub alias: String,
    pub backend: DatabaseBackend,
    pub name: String,                    // database name or SQLite file path
    pub host: String,                    // default: "localhost"
    pub port: u16,                       // default: backend-dependent
    pub user: String,
    pub password: String,
    pub pool: PoolConfig,
    pub test_name: Option<String>,
}

#[derive(Debug, Clone)]
pub struct PoolConfig {
    pub max_connections: u32,            // default: 10
    pub min_connections: u32,            // default: 0
    pub connect_timeout: Duration,       // default: 30s
    pub idle_timeout: Option<Duration>,  // default: None
    pub max_lifetime: Option<Duration>,  // default: None
    pub acquire_timeout: Duration,       // default: 30s
}

impl Default for PoolConfig {
    fn default() -> Self {
        Self {
            max_connections: 10,
            min_connections: 0,
            connect_timeout: Duration::from_secs(30),
            idle_timeout: None,
            max_lifetime: None,
            acquire_timeout: Duration::from_secs(30),
        }
    }
}

impl DatabaseConfig {
    /// Build connection URL string.
    pub fn connection_url(&self) -> String {
        match self.backend {
            DatabaseBackend::Sqlite => {
                if self.name == ":memory:" {
                    "sqlite::memory:".to_string()
                } else {
                    format!("sqlite://{}", self.name)
                }
            }
            DatabaseBackend::Mysql => format!(
                "mysql://{}:{}@{}:{}/{}",
                self.user, self.password, self.host, self.port, self.name
            ),
            DatabaseBackend::Postgres => format!(
                "postgres://{}:{}@{}:{}/{}",
                self.user, self.password, self.host, self.port, self.name
            ),
        }
    }

    /// Parse from a Python dict-equivalent HashMap.
    pub fn from_map(alias: &str, map: &HashMap<String, serde_json::Value>) -> crate::Result<Self> {
        let engine = map.get("ENGINE")
            .and_then(|v| v.as_str())
            .ok_or_else(|| OrmpiError::ConfigError("ENGINE is required".into()))?;
        let backend = DatabaseBackend::from_engine(engine)?;

        let default_port = match backend {
            DatabaseBackend::Sqlite  => 0,
            DatabaseBackend::Mysql   => 3306,
            DatabaseBackend::Postgres => 5432,
        };

        Ok(DatabaseConfig {
            alias: alias.to_string(),
            backend,
            name: str_field(map, "NAME")?,
            host: str_field_or(map, "HOST", "localhost"),
            port: u16_field_or(map, "PORT", default_port),
            user: str_field_or(map, "USER", ""),
            password: str_field_or(map, "PASSWORD", ""),
            pool: parse_pool_options(map),
            test_name: map.get("TEST")
                .and_then(|v| v.get("NAME"))
                .and_then(|v| v.as_str())
                .map(str::to_string),
        })
    }
}
```

## OrmpiPool

```rust
// ormpi-core/src/pool.rs
use sqlx::{Pool, Sqlite, MySql, Postgres};

pub enum OrmpiPool {
    Sqlite(Pool<Sqlite>),
    Mysql(Pool<MySql>),
    Postgres(Pool<Postgres>),
}

impl OrmpiPool {
    /// Create a pool from a DatabaseConfig.
    pub async fn connect(config: &DatabaseConfig) -> crate::Result<Self> {
        let url = config.connection_url();
        let pool_opts = &config.pool;

        match config.backend {
            DatabaseBackend::Sqlite => {
                let pool = sqlx::sqlite::SqlitePoolOptions::new()
                    .max_connections(pool_opts.max_connections)
                    .min_connections(pool_opts.min_connections)
                    .acquire_timeout(pool_opts.acquire_timeout)
                    .connect(&url).await?;
                Ok(OrmpiPool::Sqlite(pool))
            }
            DatabaseBackend::Mysql => {
                let pool = sqlx::mysql::MySqlPoolOptions::new()
                    .max_connections(pool_opts.max_connections)
                    .min_connections(pool_opts.min_connections)
                    .acquire_timeout(pool_opts.acquire_timeout)
                    .connect(&url).await?;
                Ok(OrmpiPool::Mysql(pool))
            }
            DatabaseBackend::Postgres => {
                let pool = sqlx::postgres::PgPoolOptions::new()
                    .max_connections(pool_opts.max_connections)
                    .min_connections(pool_opts.min_connections)
                    .acquire_timeout(pool_opts.acquire_timeout)
                    .connect(&url).await?;
                Ok(OrmpiPool::Postgres(pool))
            }
        }
    }

    pub fn backend(&self) -> DatabaseBackend {
        match self {
            OrmpiPool::Sqlite(_)   => DatabaseBackend::Sqlite,
            OrmpiPool::Mysql(_)    => DatabaseBackend::Mysql,
            OrmpiPool::Postgres(_) => DatabaseBackend::Postgres,
        }
    }

    pub async fn close(&self) {
        match self {
            OrmpiPool::Sqlite(p)   => p.close().await,
            OrmpiPool::Mysql(p)    => p.close().await,
            OrmpiPool::Postgres(p) => p.close().await,
        }
    }
}

impl std::fmt::Debug for OrmpiPool {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        match self {
            Self::Sqlite(_)   => write!(f, "OrmpiPool::Sqlite(...)"),
            Self::Mysql(_)    => write!(f, "OrmpiPool::Mysql(...)"),
            Self::Postgres(_) => write!(f, "OrmpiPool::Postgres(...)"),
        }
    }
}
```

## DatabaseRegistry

```rust
// ormpi-core/src/registry.rs
use once_cell::sync::OnceCell;
use std::collections::HashMap;
use std::sync::{Arc, RwLock};

static REGISTRY: OnceCell<Arc<RwLock<DatabaseRegistry>>> = OnceCell::new();

pub struct DatabaseRegistry {
    pub pools: HashMap<String, OrmpiPool>,
    pub configs: HashMap<String, DatabaseConfig>,
}

impl DatabaseRegistry {
    pub fn global() -> Arc<RwLock<DatabaseRegistry>> {
        REGISTRY.get_or_init(|| Arc::new(RwLock::new(DatabaseRegistry {
            pools: HashMap::new(),
            configs: HashMap::new(),
        }))).clone()
    }

    pub fn get_pool(&self, alias: &str) -> crate::Result<&OrmpiPool> {
        self.pools.get(alias).ok_or_else(|| OrmpiError::UnknownDatabase(alias.to_string()))
    }

    pub fn get_config(&self, alias: &str) -> crate::Result<&DatabaseConfig> {
        self.configs.get(alias).ok_or_else(|| OrmpiError::UnknownDatabase(alias.to_string()))
    }
}

/// Initialize all database connections from a list of configs.
pub async fn initialize(configs: Vec<DatabaseConfig>) -> crate::Result<()> {
    let mut registry = DatabaseRegistry {
        pools: HashMap::new(),
        configs: HashMap::new(),
    };

    for config in configs {
        let pool = OrmpiPool::connect(&config).await?;
        registry.pools.insert(config.alias.clone(), pool);
        registry.configs.insert(config.alias.clone(), config);
    }

    let global = DatabaseRegistry::global();
    let mut guard = global.write().unwrap();
    *guard = registry;
    Ok(())
}
```
