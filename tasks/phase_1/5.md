## Error Types

```rust
// ormpi-core/src/error.rs
use thiserror::Error;

pub type Result<T> = std::result::Result<T, OrmpiError>;

#[derive(Debug, Error)]
pub enum OrmpiError {
    // Configuration
    #[error("Unknown database backend: {0}")]
    UnknownBackend(String),

    #[error("Unknown database alias: {0}")]
    UnknownDatabase(String),

    #[error("Configuration error: {0}")]
    ConfigError(String),

    // Database / sqlx errors
    #[error("Database error: {0}")]
    Database(#[from] sqlx::Error),

    #[error("Database connection failed: {0}")]
    ConnectionFailed(String),

    // ORM errors
    #[error("{model}.DoesNotExist: {model} matching query does not exist.")]
    DoesNotExist { model: String },

    #[error("get() returned more than one {model}: {count} objects returned")]
    MultipleObjectsReturned { model: String, count: usize },

    #[error("Field error on {model}.{field}: {message}")]
    FieldError { model: String, field: String, message: String },

    #[error("Validation error: {0}")]
    ValidationError(String),

    // Transaction errors
    #[error("Transaction error: {0}")]
    TransactionError(String),

    // Migration errors
    #[error("Migration error: {0}")]
    MigrationError(String),

    #[error("Migration conflict: {0}")]
    MigrationConflict(String),

    // Query errors
    #[error("Invalid field lookup: {0}")]
    InvalidLookup(String),

    #[error("Cannot resolve keyword '{keyword}' into field. Choices are: {choices}")]
    UnknownField { keyword: String, choices: String },
}

// Convert OrmpiError → Python exceptions
impl From<OrmpiError> for pyo3::PyErr {
    fn from(err: OrmpiError) -> Self {
        use pyo3::exceptions::*;
        match &err {
            OrmpiError::DoesNotExist { .. } =>
                PyLookupError::new_err(err.to_string()),
            OrmpiError::MultipleObjectsReturned { .. } =>
                PyValueError::new_err(err.to_string()),
            OrmpiError::ValidationError(_) | OrmpiError::FieldError { .. } =>
                PyValueError::new_err(err.to_string()),
            OrmpiError::TransactionError(_) =>
                PyRuntimeError::new_err(err.to_string()),
            OrmpiError::Database(sqlx::Error::RowNotFound) =>
                PyLookupError::new_err(err.to_string()),
            _ =>
                PyRuntimeError::new_err(err.to_string()),
        }
    }
}
```

## Tokio Runtime Setup

```rust
// ormpi-py/src/runtime.rs
use once_cell::sync::OnceCell;
use tokio::runtime::Runtime;

static TOKIO_RUNTIME: OnceCell<Runtime> = OnceCell::new();

pub fn runtime() -> &'static Runtime {
    TOKIO_RUNTIME.get_or_init(|| {
        tokio::runtime::Builder::new_multi_thread()
            .worker_threads(4)
            .enable_all()
            .build()
            .expect("Failed to create Tokio runtime")
    })
}

/// Block on a future using the global Tokio runtime.
/// Used only in CLI context where we cannot be async.
pub fn block_on<F: std::future::Future>(f: F) -> F::Output {
    runtime().block_on(f)
}
```

## Async Bridge Pattern in ormpi-py

Every async Rust function exposed to Python follows this pattern:

```rust
// Pattern for exposing async fn to Python
#[pyfunction]
fn my_async_fn(py: Python<'_>, arg: String) -> PyResult<&PyAny> {
    pyo3_async_runtimes::tokio::future_into_py(py, async move {
        // Run async Rust code on the tokio runtime
        let result = ormpi_core::do_something(&arg).await
            .map_err(|e| PyErr::from(e))?;
        Ok(Python::with_gil(|py| result.into_py(py)))
    })
}
```

Key rules:
1. Always release the GIL during the `async move` block — don't hold Python objects unless needed.
2. Use `Python::with_gil(|py| ...)` only when converting results back.
3. Never call blocking I/O inside the async block.

// ormpi-core/src/executor.rs
use sqlx::Row;

pub async fn fetch_one_raw(
    pool: &OrmpiPool,
    sql: &str,
) -> crate::Result<Vec<(String, serde_json::Value)>> {
    match pool {
        OrmpiPool::Sqlite(p) => {
            let row = sqlx::query(sql).fetch_one(p).await?;
            // Convert row columns to JSON values
            Ok(row_to_json_sqlite(&row))
        }
        OrmpiPool::Mysql(p) => {
            let row = sqlx::query(sql).fetch_one(p).await?;
            Ok(row_to_json_mysql(&row))
        }
        OrmpiPool::Postgres(p) => {
            let row = sqlx::query(sql).fetch_one(p).await?;
            Ok(row_to_json_postgres(&row))
        }
    }
}
