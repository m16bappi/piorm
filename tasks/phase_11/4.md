# Task 11.4 — Auto-Sync Tables on `configure()` (syncdb Mode)

## Objective

Add an `auto_sync: bool = False` parameter to `ormpi.configure()`. When set to `True`,
ormpi automatically runs `CREATE TABLE IF NOT EXISTS` (and junction tables for
`ManyToManyField`) for every registered model immediately after the connection pools are
ready. No migration files are read or written. No CLI command needs to be run.

This makes ormpi usable for small projects and scripts exactly like:
- Django's old `syncdb` / `migrate --run-syncdb`
- SQLAlchemy's `Base.metadata.create_all()`
- Tortoise ORM's `await Tortoise.generate_schemas()`

---

## User-Facing API

```python
import ormpi
from ormpi import Model, CharField, IntegerField

class User(Model):
    name  = CharField(max_length=100)
    score = IntegerField(default=0)

async def main():
    await ormpi.configure(
        {"default": {"ENGINE": "sqlite", "NAME": "app.db"}},
        auto_sync=True,   # ← single flag; no migrations needed
    )
    # Tables exist now. Ready to use.
    user = await User.objects.create(name="Alice", score=42)
    print(user.pk, user.name)

asyncio.run(main())
```

For production projects that want migration control, leave `auto_sync=False` (default)
and use the CLI workflow as documented in phases 7–8.

## Rust: Updated `configure()` Signature

```rust
// ormpi-py/src/lib.rs

#[pyfunction]
#[pyo3(signature = (databases, auto_sync = false))]
pub fn configure(
    py: Python<'_>,
    databases: &PyDict,
    auto_sync: bool,
) -> PyResult<PyObject> {
    let config = parse_database_config(py, databases)?;

    pyo3_async_runtimes::tokio::future_into_py(py, async move {
        // 1. Initialize pools (existing logic from phase 5)
        initialize_pools(config).await?;

        // 2. Optionally sync tables
        if auto_sync {
            sync_all_tables().await?;
        }

        Ok(())
    })
}
```

## Rust: `sync_all_tables()` Implementation

```rust
// ormpi-core/src/sync.rs

use crate::models::ModelRegistry;
use crate::registry::DatabaseRegistry;
use crate::ddl::model_to_ddl;
use crate::error::OrmpiError;

/// Create all registered model tables using CREATE TABLE IF NOT EXISTS.
/// Also creates M2M junction tables.
/// Safe to call multiple times — idempotent.
pub async fn sync_all_tables() -> Result<(), OrmpiError> {
    let registry     = ModelRegistry::global().read().unwrap();
    let db_registry  = DatabaseRegistry::global().read().unwrap();

    for (model_name, model_def) in registry.iter() {
        // Skip abstract models — they have no table
        if model_def.meta.abstract_ { continue; }

        // Determine which database alias to use for this model
        // (respects db routing if configured, otherwise "default")
        let db_alias = db_registry.write_alias_for(model_name).unwrap_or("default");
        let pool     = db_registry.get_pool(db_alias)?;
        let backend  = pool.backend();

        // Generate CREATE TABLE IF NOT EXISTS for the model
        let stmts = model_to_ddl_if_not_exists(model_def, backend)?;

        for sql in stmts {
            pool.execute_raw(&sql, &[]).await?;
        }
    }

    Ok(())
}
```

## Rust: `model_to_ddl_if_not_exists()`

The existing `model_to_ddl()` (from phase 3 task 3) generates a plain `CREATE TABLE`.
Add a variant that uses `CREATE TABLE IF NOT EXISTS`:

```rust
// ormpi-core/src/ddl.rs

use sea_query::{TableCreateStatement, IfNotExists};

pub fn model_to_ddl_if_not_exists(
    model_def: &ModelDef,
    backend: DatabaseBackend,
) -> Result<Vec<String>, OrmpiError> {
    let mut stmts = Vec::new();

    // ── Main table ───────────────────────────────────────────────────────────
    let table_name = model_def.table_name();
    let mut create = TableCreateStatement::new();
    create
        .table(Alias::new(&table_name))
        .if_not_exists();                     // ← key difference

    for field in &model_def.fields {
        // Skip M2M fields — they get a junction table, not a column
        if matches!(field.field_type, FieldType::ManyToMany(_)) { continue; }

        let col_def = field.to_column_def(backend);
        create.col(col_def);
    }

    // Foreign key constraints (not on SQLite — enforced via PRAGMA)
    if !matches!(backend, DatabaseBackend::Sqlite) {
        for fk in &model_def.foreign_keys() {
            create.foreign_key(build_fk_constraint(fk, &table_name));
        }
    }

    let sql = match backend {
        DatabaseBackend::Sqlite   => create.to_string(SqliteQueryBuilder),
        DatabaseBackend::Mysql    => create.to_string(MysqlQueryBuilder),
        DatabaseBackend::Postgres => create.to_string(PostgresQueryBuilder),
    };
    stmts.push(sql);

    // ── Indexes ──────────────────────────────────────────────────────────────
    for field in &model_def.fields {
        if field.db_index && !field.primary_key {
            let idx_name = format!("idx_{}_{}", table_name, field.attname);
            let idx_sql = build_index_sql(&table_name, &field.db_column(), &idx_name, backend);
            stmts.push(format!("CREATE INDEX IF NOT EXISTS {} ON {} ({})",
                idx_name, table_name, field.db_column()));
        }
    }

    // ── M2M junction tables ──────────────────────────────────────────────────
    for m2m in model_def.m2m_fields() {
        let junction_name = m2m.through_table.clone()
            .unwrap_or_else(|| junction_table_name(&table_name, &m2m.related_table));

        let mut jct = TableCreateStatement::new();
        jct.table(Alias::new(&junction_name))
           .if_not_exists()
           .col(ColumnDef::new(Alias::new("id"))
               .big_integer()
               .not_null()
               .auto_increment()
               .primary_key())
           .col(ColumnDef::new(Alias::new(&format!("{}_id", table_name)))
               .big_integer()
               .not_null())
           .col(ColumnDef::new(Alias::new(&format!("{}_id", m2m.related_table)))
               .big_integer()
               .not_null());

        let jct_sql = match backend {
            DatabaseBackend::Sqlite   => jct.to_string(SqliteQueryBuilder),
            DatabaseBackend::Mysql    => jct.to_string(MysqlQueryBuilder),
            DatabaseBackend::Postgres => jct.to_string(PostgresQueryBuilder),
        };
        stmts.push(jct_sql);
    }

    Ok(stmts)
}

fn junction_table_name(a: &str, b: &str) -> String {
    // Alphabetical to ensure stability regardless of declaration order
    let (first, second) = if a < b { (a, b) } else { (b, a) };
    format!("{}_{}", first, second)
}
```

### Why `IF NOT EXISTS` is the right strategy for auto-sync

- **Idempotent** — calling `configure(auto_sync=True)` multiple times is safe.
- **Non-destructive** — existing tables with real data are never touched.
- **No schema diffing needed** — this is not a migration; column changes to existing tables
  are not applied. For schema evolution, use migrations. Auto-sync only ensures tables exist.

This is exactly the same semantic as SQLAlchemy's `checkfirst=True` (the default).

## Ordering: Models Must Be Registered Before `configure()`

Auto-sync works because `Model.__init_subclass__` registers every model class at import
time. The user must import their models before calling `configure()`:

```python
# Correct — models imported before configure
from myapp.models import User, Post, Comment

await ormpi.configure({"default": {...}}, auto_sync=True)

# Incorrect — configure called before models are imported
await ormpi.configure({"default": {...}}, auto_sync=True)
from myapp.models import User   # too late — User not in registry during sync
```

To document this, add a check and warning in `sync_all_tables()`:

```rust
if registry.is_empty() {
    // Emit a Python warning rather than silently doing nothing
    Python::with_gil(|py| {
        py.run(
            "import warnings; warnings.warn(
                'auto_sync=True was set but no models are registered. '
                'Import your models before calling configure().',
                stacklevel=3
            )",
            None, None
        )
    }).ok();
}
```

## Integration Tests

```python
# tests/integration/test_auto_sync.py
import pytest
import ormpi
from ormpi import Model, CharField, IntegerField, ForeignKey, CASCADE

class Widget(Model):
    name  = CharField(max_length=100)
    count = IntegerField(default=0)

class Gadget(Model):
    name   = CharField(max_length=100)
    widget = ForeignKey(Widget, on_delete=CASCADE)

async def test_auto_sync_creates_tables():
    """Tables must not exist before configure, exist after."""
    await ormpi.configure(
        {"default": {"ENGINE": "sqlite", "NAME": ":memory:"}},
        auto_sync=True,
    )
    # If tables didn't exist, this would raise OperationalError
    widget = await Widget.objects.create(name="Cog", count=3)
    gadget = await Gadget.objects.create(name="Gizmo", widget=widget)
    assert gadget.pk is not None
    await ormpi.close()

async def test_auto_sync_idempotent():
    """Calling configure with auto_sync=True twice must not raise."""
    config = {"default": {"ENGINE": "sqlite", "NAME": ":memory:"}}
    await ormpi.configure(config, auto_sync=True)
    await ormpi.close()
    await ormpi.configure(config, auto_sync=True)   # second call — IF NOT EXISTS
    result = await Widget.objects.count()
    assert result == 0
    await ormpi.close()

async def test_auto_sync_false_does_not_create():
    """auto_sync=False (default) must not create tables."""
    await ormpi.configure(
        {"default": {"ENGINE": "sqlite", "NAME": ":memory:"}},
        auto_sync=False,
    )
    with pytest.raises(ormpi.OperationalError):
        await Widget.objects.count()   # table does not exist
    await ormpi.close()

async def test_auto_sync_with_m2m():
    """Junction tables for M2M fields must also be created."""
    from ormpi import ManyToManyField

    class Category(Model):
        name = CharField(max_length=50)

    class Article(Model):
        title      = CharField(max_length=200)
        categories = ManyToManyField(Category)

    await ormpi.configure(
        {"default": {"ENGINE": "sqlite", "NAME": ":memory:"}},
        auto_sync=True,
    )
    cat = await Category.objects.create(name="Tech")
    art = await Article.objects.create(title="Hello")
    await art.categories.add(cat)
    cats = await art.categories.all()
    assert len(cats) == 1
    await ormpi.close()
```
